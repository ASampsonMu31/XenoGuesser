#version 330 core

// Window glass: mostly clear, with the sky reflected in it, more strongly at a glancing
// angle, and a bright glint of sun

in vec3 aPos;
in vec3 aNormal;
out vec4 fragColour;

uniform vec3 viewPos;
uniform vec3 sunPos;
uniform vec3 sunColour;
uniform vec3 ambientLight;
uniform vec3 glassTint;
uniform sampler2D skyTexture;
uniform mat4 skyRotation;

const float PI = 3.14159265359;

vec3 skyAlong(vec3 dir) {
    vec3 local = transpose(mat3(skyRotation)) * normalize(dir);
    float theta = atan(local.x, local.z);
    if (theta < 0.0) theta += 2.0 * PI;
    return texture(skyTexture, vec2(theta / (2.0 * PI), asin(clamp(local.y, -1.0, 1.0)) / PI + 0.5)).rgb;
}

void main() {
    vec3 n = normalize(aNormal);
    vec3 viewDir = normalize(viewPos - aPos);
    if (dot(n, viewDir) < 0.0) n = -n;
    float facing = max(dot(n, viewDir), 0.0);
    float fresnel = 0.04 + 0.96 * pow(1.0 - facing, 5.0);

    vec3 reflected = skyAlong(reflect(-viewDir, n));
    vec3 lightDir = normalize(sunPos - aPos);
    float glint = pow(max(dot(n, normalize(lightDir + viewDir)), 0.0), 220.0) * 1.6;

    float reflectAmount = 0.45 + 0.5 * fresnel;
    vec3 colour = mix(glassTint * (ambientLight * 1.3 + sunColour * 0.2), reflected, reflectAmount) + sunColour * glint;
    float alpha = clamp(0.3 + 0.5 * fresnel + glint, 0.0, 0.92);

    // Fading into the distance like everything else
    float distanceAway = length(viewPos - aPos);
    float fog = smoothstep(0.0, 1.0, clamp((2200.0 - distanceAway) / 1200.0, 0.0, 1.0));
    fragColour = vec4(colour, alpha * fog);
}
