#version 330 core

// Organisms: each vertex is moved by the world matrix of the body part (bone) it belongs to

layout (location = 0) in vec3 position;
layout (location = 1) in vec3 normal;
layout (location = 2) in vec2 texCoord;
layout (location = 3) in float boneIndex;
layout (location = 4) in float partKind;

out vec3 aPos;
out vec3 aNormal;
out vec2 vUV;
flat out int vPart;

uniform mat4 bones[48];
uniform mat4 viewProjection;
// Takes the posed body, in its own frame (x east, y up, z south about where it stands), into
// the world less the camera's position (SphereFrame.modelMatrix(scale, eye), worked out in
// double precision); the identity for something drawn in its own space (a picture, the
// player's body in the camera's frame)
uniform mat4 anchor;

void main() {
    mat4 bone = anchor * bones[int(boneIndex + 0.5)];
    vec4 world = bone * vec4(position, 1.0);
    aPos = world.xyz;
    aNormal = normalize(mat3(bone) * normal);
    gl_Position = viewProjection * world;
    vUV = texCoord;
    vPart = int(partKind + 0.5);
}
