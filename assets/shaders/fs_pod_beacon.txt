#version 330 core

// Bright green light, nearly white at its hottest middle, in a softer green glow
in vec3 place;
out vec4 fragColour;

void main() {
    float out2 = dot(place.xy, place.xy);
    float core = exp(-out2 * 14.0);
    float amount = clamp((core + 0.4 * exp(-out2 * 3.0)) * place.z, 0.0, 1.0);
    vec3 colour = mix(vec3(0.15, 1.0, 0.25), vec3(0.8, 1.0, 0.8), core * place.z * 0.6);
    // (premultiplied by how much of it there is)
    fragColour = vec4(colour * amount, amount);
}
