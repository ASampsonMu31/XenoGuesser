#version 330 core

// White light, a solid bright middle edged by a fringe in the sky's complementary colour
// (darkened): the white stands out against a dark or coloured sky, the fringe against a pale or
// clouded one, and by its hue against any
in vec3 place;
out vec4 fragColour;
uniform vec3 fringeColour;

void main() {
    float out2 = dot(place.xy, place.xy);
    float core = exp(-out2 * 8.0);
    // (gone before the light's edges, so the ball's square and the beam's sides never show)
    float fringe = max(0.0, exp(-out2 * 1.5) - core) * (1.0 - smoothstep(0.45, 1.0, out2));
    float white = clamp(1.4 * core * place.z, 0.0, 1.0);
    float dark = clamp(0.7 * fringe * place.z, 0.0, 1.0) * (1.0 - white);
    // (premultiplied: the white over the fringe over whatever's behind)
    fragColour = vec4(vec3(white) + fringeColour * dark, white + dark);
}
