#version 330 core

in vec3 viewPosition;
in vec3 surfaceNormal;
in float waveHeight;

uniform vec3 waterColour;
uniform vec3 skyColour;
uniform vec3 sunDirection;
uniform float daylight;

out vec4 fragColour;

void main() {
    vec3 n = normalize(surfaceNormal);
    vec3 viewDir = normalize(-viewPosition);
    float diffuse = max(dot(n, normalize(sunDirection)), 0.0);
    float fresnel = pow(1.0 - max(dot(n, viewDir), 0.0), 4.0);
    float glint = pow(max(dot(reflect(-normalize(sunDirection), n), viewDir), 0.0), 96.0);
    vec3 deep = waterColour * (0.58 + 0.35 * diffuse * daylight);
    vec3 reflected = mix(skyColour * 0.65, waterColour * 1.25, 0.35 + 0.25 * daylight);
    vec3 colour = mix(deep, reflected, clamp(fresnel * 0.72, 0.0, 0.72));
    colour += vec3(0.8, 0.88, 1.0) * glint * daylight * (0.65 + 0.15 * waveHeight);
    fragColour = vec4(colour, 0.94);
}
