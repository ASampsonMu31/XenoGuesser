#version 330 core

// The sea's surface (see vs_sphere_water): see-through where it is shallow over the land,
// rippled with the sea texture, the sky reflected in it, the sun glittering on it, whitecaps
// on big waves and foam lapping along the shore. Everything planet-centred is in world
// directions; up is the planet's up where the fragment is.

// What the distance fades into: an even haze, a greyed version of the sky's colour (greyer
// still under cloud), the same in every direction (see Shader.setFog)
uniform vec3 fogColour;
uniform float fogReady;
vec3 fogHaze(vec3 fallback) {
    return fogReady > 0.5 ? fogColour : fallback;
}

in vec3 viewPosition;
in vec3 worldPosition;
in vec3 aNormal;
in float PlanarDepth;
in float vCrest;
in float vWaveStrength;

out vec4 fragColour;

uniform vec2 windowSize;

uniform sampler2D diffuse_texture;
uniform sampler2D terrainDepthTexture;
uniform sampler2D skyTexture;

// The direction to the sun, in the world
uniform vec3 sunDirection;
uniform vec3 lightSpecular;
uniform vec3 matSpecular;
uniform float matShininess;

uniform float time;
uniform mat4 skyRotation;

// Seeded per world: colour seen through shallow water, and of open deep ocean
uniform vec3 u_ShallowTint;
uniform vec3 u_DeepOceanColour;

const float SHALLOW_OPACITY_FLOOR = 0.28;
const float EDGE_FADE_DISTANCE    = 0.5;
const float DEPTH_FALLOFF_SPEED   = 0.20;
const float RIPPLE_SPEED          = 0.11;
const float RIPPLE_STRENGTH       = 16.0;
const float FINE_RIPPLE_SCALE     = 0.034;
const float FINE_RIPPLE_SPEED     = 0.23;
const float SKY_REFLECTION        = 0.5;
const float PI                    = 3.14159265359;

// Must match the scene projection, which pulls its near plane in while swimming
uniform float nearPlane;
uniform float farPlane;

vec3 skyAlong(vec3 worldDir) {
    vec3 localSkyDir = transpose(mat3(skyRotation)) * worldDir;
    float theta = atan(localSkyDir.x, localSkyDir.z);
    if (theta < 0.0) {
        theta += 2.0 * PI;
    }
    return texture(skyTexture, vec2(theta / (2.0 * PI), asin(clamp(localSkyDir.y, -1.0, 1.0)) / PI + 0.5)).rgb;
}

float linearizeDepth(float rawDepth) {
    float zNDC = rawDepth * 2.0 - 1.0;
    return (2.0 * nearPlane * farPlane) / (farPlane + nearPlane - zNDC * (farPlane - nearPlane));
}

// The ripples on one flat projection of the surface, p in world units: the four drifting
// layers combined, two of them multiplied (for the foam's pattern), and the finer set
struct Ripples {
    vec3 combined;
    vec3 foam;
    float fine;
};

Ripples ripplesAt(vec2 p) {
    vec3 tex1 = texture(diffuse_texture, p * 0.010 + vec2(0.0, time * RIPPLE_SPEED)).rgb;
    vec3 tex2 = texture(diffuse_texture, p * 0.011 + vec2(0.0, -time * RIPPLE_SPEED * 1.01)).rgb;
    vec3 tex3 = texture(diffuse_texture, p * 0.009 + vec2(time * RIPPLE_SPEED * 1.02, 0.0)).rgb;
    vec3 tex4 = texture(diffuse_texture, p * 0.010 + vec2(-time * RIPPLE_SPEED * 1.03, 0.0)).rgb;
    // A finer, quicker set of ripples drifting across the wind
    vec3 fine1 = texture(diffuse_texture, p * FINE_RIPPLE_SCALE + vec2(time * FINE_RIPPLE_SPEED, time * 0.07)).rgb;
    vec3 fine2 = texture(diffuse_texture, p * FINE_RIPPLE_SCALE * 1.37 + vec2(-time * 0.09, -time * FINE_RIPPLE_SPEED)).rgb;
    Ripples r;
    r.combined = tex1 * tex2 + tex3 * tex4;
    r.foam = tex1 * tex3;
    r.fine = dot(fine1 * fine2, vec3(0.333));
    return r;
}

void main() {
    vec3 up = normalize(worldPosition);
    vec3 east = vec3(up.z, 0.0, -up.x);
    east = length(east) > 1e-6 ? normalize(east) : vec3(1.0, 0.0, 0.0);
    vec3 south = cross(east, up);
    vec3 viewDir = normalize(-viewPosition);
    vec2 screenUV = gl_FragCoord.xy / windowSize;

    float rawDepth = texture(terrainDepthTexture, screenUV).r;
    float landDistance = linearizeDepth(rawDepth);
    float waterDistance = PlanarDepth;

    float totalThickness = max(landDistance - waterDistance, 0.0);

    float cosTheta = max(abs(dot(viewDir, up)), 0.0001);

    float actualRayPathLength = totalThickness / cosTheta;
    float stretchedPathLength = actualRayPathLength * DEPTH_FALLOFF_SPEED;

    vec3 absorptionCoefficients = vec3(0.015, 0.005, 0.001);
    vec3 transparencyFactor = exp(-stretchedPathLength * absorptionCoefficients);

    // The ripple texture laid over the sphere from the three axis planes, each weighted by
    // how squarely the surface faces along its axis (no seams, nothing stretched)
    vec3 weights = pow(abs(up), vec3(4.0));
    weights /= weights.x + weights.y + weights.z;
    Ripples ripples;
    ripples.combined = vec3(0.0);
    ripples.foam = vec3(0.0);
    ripples.fine = 0.0;
    if (weights.x > 0.01) {
        Ripples r = ripplesAt(worldPosition.zy);
        ripples.combined += r.combined * weights.x; ripples.foam += r.foam * weights.x; ripples.fine += r.fine * weights.x;
    }
    if (weights.y > 0.01) {
        Ripples r = ripplesAt(worldPosition.xz);
        ripples.combined += r.combined * weights.y; ripples.foam += r.foam * weights.y; ripples.fine += r.fine * weights.y;
    }
    if (weights.z > 0.01) {
        Ripples r = ripplesAt(worldPosition.xy);
        ripples.combined += r.combined * weights.z; ripples.foam += r.foam * weights.z; ripples.fine += r.fine * weights.z;
    }
    vec3 combinedTex = ripples.combined;
    float fineRipples = ripples.fine;
    vec3 waterSurfaceTex = clamp(combinedTex * 1.5, 0.0, 1.0);

    float factor = transparencyFactor.g;
    vec3 shallowWaterBlend = mix(u_ShallowTint, waterSurfaceTex, factor);
    vec3 waterBaseColour = mix(u_DeepOceanColour, shallowWaterBlend, factor);

    float edgeFade = clamp(totalThickness / EDGE_FADE_DISTANCE, 0.0, 1.0);
    float baseOpacity = mix(SHALLOW_OPACITY_FLOOR, 1.0, 1.0 - factor);
    float opacity = baseOpacity * edgeFade;

    // Ripple normals: the slope of the blended surface pattern across the ground (east,
    // south), from screen-space derivatives, so the sun's reflection breaks up across the ripples
    float rippleHeight = dot(combinedTex, vec3(0.333)) + 0.6 * fineRipples;
    vec2 heightDerivs = vec2(dFdx(rippleHeight), dFdy(rippleHeight));
    vec3 stepX = dFdx(viewPosition), stepY = dFdy(viewPosition);
    mat2 groundDerivs = mat2(dot(stepX, east), dot(stepX, south), dot(stepY, east), dot(stepY, south));
    float det = determinant(groundDerivs);
    vec2 slope = abs(det) > 1e-6 ? inverse(transpose(groundDerivs)) * heightDerivs : vec2(0.0);
    // Ripples fade into the distance, where they would only shimmer, and ride on the waves
    float rippleFade = 1.0 - smoothstep(60.0, 480.0, length(viewPosition));
    vec2 rippleTilt = -slope * RIPPLE_STRENGTH * rippleFade;
    vec3 waveNormal = normalize(aNormal);
    vec3 normal = normalize(waveNormal + east * rippleTilt.x + south * rippleTilt.y);
    vec3 lightDir = normalize(sunDirection);
    vec3 halfwayDir = normalize(lightDir + viewDir);

    float facing = max(dot(normal, halfwayDir), 0.0);
    // A tight, bright core plus a broader glitter that the ripples scatter across the water
    vec3 specularHighlight = lightSpecular * (matSpecular * pow(facing, matShininess) + 0.9 * pow(facing, 160.0));
    // (none from a sun below the horizon)
    specularHighlight *= smoothstep(-0.05, 0.05, dot(lightDir, up));

    // Wave faces turned to the sun are lighter, those turned away darker
    float waveLight = dot(waveNormal, lightDir);
    waterBaseColour *= 0.82 + 0.3 * waveLight;

    // The sky reflected in the surface, strongest at glancing angles; this is what makes the
    // ripples visible everywhere, not just in the sun's glitter
    vec3 reflected = reflect(-viewDir, normal);
    float below = dot(reflected, up);
    if (below < 0.0) reflected -= 2.0 * below * up;
    float fresnel = 0.04 + 0.96 * pow(1.0 - max(dot(normal, viewDir), 0.0), 5.0);
    waterBaseColour = mix(waterBaseColour, skyAlong(reflected), clamp(fresnel * SKY_REFLECTION, 0.0, SKY_REFLECTION));

    // Foam: whitecaps on the crests of big waves, and a lapping line along the shore
    vec3 foamColour = mix(vec3(0.92), u_ShallowTint, 0.15);
    float foamPattern = smoothstep(0.35, 0.75, dot(ripples.foam, vec3(0.333)) * 2.2 + fineRipples * 0.6);
    float whitecaps = smoothstep(0.78, 0.97, vCrest) * smoothstep(0.4, 0.9, vWaveStrength);
    float shoreBand = 1.0 - smoothstep(0.6, 4.5, totalThickness);
    float lapping = 0.5 + 0.5 * sin(totalThickness * 1.6 - time * 1.8 + fineRipples * 5.0);
    float foam = clamp(whitecaps * foamPattern + shoreBand * lapping * foamPattern * 0.8, 0.0, 1.0);
    waterBaseColour = mix(waterBaseColour, foamColour, foam);
    opacity = max(opacity, foam * edgeFade);

    vec3 baseWaterColourWithSpecular = waterBaseColour + specularHighlight * (1.0 - foam);

    // Into the distance it fades to the haze (or the sky behind it)
    float radialDistance = length(viewPosition);
    // (the sky's own colour is only the haze until the fog's colour is set: not worked out otherwise)
    vec3 dynamicSkyColour = fogReady > 0.5 ? fogColour : skyAlong(normalize(viewPosition));
    float fogStart = 1000.0;
    float fogEnd = 2200.0;
    float fogFactor = clamp((fogEnd - radialDistance) / (fogEnd - fogStart), 0.0, 1.0);
    fogFactor = smoothstep(0.0, 1.0, fogFactor);

    vec3 finalColourWithFog = mix(fogHaze(dynamicSkyColour), baseWaterColourWithSpecular, fogFactor);

    // As the fog thickens with distance the water turns solid, so the haze-coloured far
    // water doesn't show the land through it as cutouts
    float finalOpacity = mix(1.0, opacity, fogFactor);

    fragColour = vec4(finalColourWithFog, finalOpacity);
}
