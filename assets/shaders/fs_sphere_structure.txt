#version 330 core

in vec3 aPos;
in vec3 aNormal;
in vec2 aTexCoord;
in float PlanarDepth;

struct Material {
    vec3 ambient;
    vec3 diffuse;
    vec3 specular;
    float shininess;
};

uniform Material material;
uniform vec3 ambientLight;
uniform vec3 viewPos;
uniform sampler2D diffuse_texture;
uniform vec2 scale;
uniform int useDiffuseTexture;
out vec4 fragColour;

void main() {
    vec3 normal = normalize(aNormal);
    vec3 radial = normalize(aPos - viewPos);
    float facing = abs(dot(normal, radial));
    float light = 0.42 + 0.58 * facing;
    vec3 albedo = useDiffuseTexture != 0 ? texture(diffuse_texture, aTexCoord * scale).rgb : vec3(1.0);
    vec3 colour = material.diffuse * albedo * (ambientLight + vec3(light));
    fragColour = vec4(colour, 1.0);
}
