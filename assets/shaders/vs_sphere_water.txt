#version 330 core

// The sea: a grid round the viewer (x, z from -1/2 to 1/2, densest in the middle), laid along
// the ground in the viewer's east/south frame and wrapped onto the sea's sphere, then displaced
// by Gerstner waves whose size comes from the wave map (nothing on small lakes, rolling swell
// on the open ocean). Each vertex works in its own east/up/south frame, so the waves are the
// same at every latitude. Drawn camera-relative (sphereCamera); the land hides it where higher.

layout (location = 0) in vec3 position;
layout (location = 1) in vec3 normal;
layout (location = 2) in vec2 texCoord;

out vec3 viewPosition;
out vec3 worldPosition;
out vec3 aNormal;
out float PlanarDepth;
out float vCrest;
out float vWaveStrength;

uniform mat4 viewProjection;
uniform vec3 sphereCamera;
// The frame on the ground under the eyes, the eyes' height above the sea, and how far the grid
// reaches along the ground from one side to the other
uniform vec3 centreUp;
uniform vec3 centreEast;
uniform float eyeAboveSea;
uniform vec3 centreSouth;
uniform float gridSize;
uniform float planetRadius;
uniform float seaLevel;
uniform float time;

// Red: how big the waves are here; green: distance to the nearest shore in world units
// (RegionalGenerationManager.waveFieldTexels), looked up by direction; waveTexel its spacing
uniform samplerCube waveMap;
uniform float waveTexel;

// World units: the viewer's eyes are about 20 above the ground
const float GRAVITY = 140.0;
const float SWELL_HEIGHT = 7.0;
const float CHOP_HEIGHT = 2.2;
const int WAVE_COUNT = 6;

// Wavelength, share of its set's height, set (0 swell, 1 chop), how much its crests wander
// (RegionalGenerationManager.WAVES must match)
const vec4 WAVES[WAVE_COUNT] = vec4[](
    vec4(620.0, 1.00, 0.0, 1.2),
    vec4(410.0, 0.60, 0.0, 2.0),
    vec4(150.0, 1.00, 1.0, 2.5),
    vec4(108.0, 0.70, 1.0, 3.5),
    vec4( 84.0, 0.55, 1.0, 4.5),
    vec4(128.0, 0.60, 1.0, 3.0)
);

float hash(vec3 p) {
    p = fract(p * 0.3183099 + 0.1);
    p *= 17.0;
    return fract(p.x * p.y * p.z * (p.x + p.y + p.z));
}

float valueNoise(vec3 p) {
    vec3 i = floor(p), f = fract(p);
    f = f * f * (3.0 - 2.0 * f);
    return mix(mix(mix(hash(i), hash(i + vec3(1, 0, 0)), f.x), mix(hash(i + vec3(0, 1, 0)), hash(i + vec3(1, 1, 0)), f.x), f.y),
               mix(mix(hash(i + vec3(0, 0, 1)), hash(i + vec3(1, 0, 1)), f.x), mix(hash(i + vec3(0, 1, 1)), hash(i + vec3(1, 1, 1)), f.x), f.y), f.z);
}

vec2 sampleMap(vec3 direction) {
    return textureLod(waveMap, direction, 0.0).rg;
}

void main() {
    // Where the grid point lies on the sea: its (x, z) taken as distances along the ground
    // from under the eyes (as SphereFrame.toSphere), kept relative to the eyes for precision
    vec2 along = position.xz * gridSize;
    float d = length(along);
    float r = planetRadius + seaLevel;
    float a = d / r;
    vec3 across = d > 0.0 ? (centreEast * along.x + centreSouth * along.y) / d : vec3(0.0);
    float halfSine = sin(a * 0.5);
    vec3 fromEyes = centreUp * (-2.0 * r * halfSine * halfSine - eyeAboveSea) + across * (r * sin(a));
    vec3 up = normalize(centreUp * cos(a) + across * sin(a));
    vec3 world = up * r;
    // The point's own frame (D10: east = the polar axis cross up, south = east cross up)
    vec3 east = vec3(up.z, 0.0, -up.x);
    east = length(east) > 1e-6 ? normalize(east) : vec3(1.0, 0.0, 0.0);
    vec3 south = cross(east, up);

    vec2 map = sampleMap(up);
    float strength = map.r;
    float shoreDistance = map.g;

    // Waves travel down the slope of the distance to shore, so their crests run along the
    // coast and they always roll in towards land
    // (measured over a few texels: where waves from two coasts meet, the slopes either side
    // point opposite ways, and over this span they cancel, so the waves die away across the
    // meeting line rather than being pushed apart along it and tearing the surface)
    float h = waveTexel * 4.0;
    vec2 slope = vec2(sampleMap(world + east * h).g - sampleMap(world - east * h).g,
                      sampleMap(world + south * h).g - sampleMap(world - south * h).g) / (2.0 * h);
    float slopeLength = length(slope);
    vec2 dir = slopeLength > 1e-4 ? -slope / slopeLength : vec2(1.0, 0.0);
    // Where waves from two coasts meet the distance has no clear slope; they die away there
    float coherence = smoothstep(0.15, 0.55, slopeLength);

    // Small waves alias between the sparse far vertices, so they fade out with distance
    float distanceToEye = length(fromEyes);
    float chopFade = 1.0 - smoothstep(500.0, 1300.0, distanceToEye);
    float pushFade = 1.0 - smoothstep(250.0, 700.0, distanceToEye);
    float swellHeight = SWELL_HEIGHT * smoothstep(0.55, 1.0, strength) * coherence;
    float chopHeight = CHOP_HEIGHT * smoothstep(0.0, 0.45, strength) * chopFade * mix(0.5, 1.0, coherence);

    // Worked in the frame: (x east, y up, z south)
    vec3 offset = vec3(0.0);
    vec3 n = vec3(0.0, 1.0, 0.0);
    float height = 0.0;
    float maxHeight = 0.0001;
    for (int i = 0; i < WAVE_COUNT; i++) {
        vec4 wave = WAVES[i];
        float amplitude = wave.y * (wave.z < 0.5 ? swellHeight : chopHeight);
        float k = 6.2831853 / wave.x;
        float omega = sqrt(GRAVITY * k);
        // Steepness shared out so the crests sharpen without folding over
        float q = amplitude > 0.0 ? min(1.0, 0.55 / (k * amplitude * float(WAVE_COUNT))) : 0.0;
        // Crests wander gently so they aren't perfectly parallel to the shore
        // (Not right round the viewer, so a swimmer's bobbing can follow the waves exactly:
        // see RegionalGenerationManager.waveSurfaceAt)
        float wander = wave.w * valueNoise(world / (900.0 + 230.0 * float(i)) + float(i) * 13.1)
                * smoothstep(40.0, 260.0, distanceToEye);
        float phase = -k * shoreDistance + wander - omega * time + float(i) * 1.7;
        float c = cos(phase), s = sin(phase);
        // (the sideways push that sharpens the crests fades out with distance: where the grid is
        // coarse it would fold neighbouring triangles over each other into a sawtooth)
        offset.xz += q * amplitude * dir * c * coherence * pushFade;
        offset.y += amplitude * s;
        n.x -= dir.x * k * amplitude * c;
        n.z -= dir.y * k * amplitude * c;
        n.y -= q * k * amplitude * s;
        height += amplitude * s;
        maxHeight += amplitude;
    }

    vec3 shift = east * offset.x + up * offset.y + south * offset.z;
    viewPosition = fromEyes + shift;
    worldPosition = world + shift;
    gl_Position = viewProjection * vec4(viewPosition, 1.0);
    PlanarDepth = gl_Position.w;
    aNormal = normalize(east * n.x + up * n.y + south * n.z);
    vCrest = height / maxHeight * 0.5 + 0.5;
    vWaveStrength = strength;
}
