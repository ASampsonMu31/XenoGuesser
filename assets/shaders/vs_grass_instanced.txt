#version 330 core

layout (location = 0) in vec3 aPos;
layout (location = 1) in vec2 aTexCoords;
// Where the blade stands on the planet (planet-centred), its colour and its height
layout (location = 2) in vec3 aInstancePos;
layout (location = 3) in float aColourPhenotype; // Driven directly from CPU
layout (location = 4) in float aBladeHeight;     // Driven directly from CPU

out vec2 TexCoords;
out vec3 fWorldPos;
out float fColourPhenotype;
out float fBladeShade;

// The camera-relative view-projection, and where the camera is (planet-centred): the blades
// are drawn relative to it
uniform mat4 viewProjection;
uniform vec3 sphereCamera;
uniform float time;
uniform int worldSeed;

// An exact hash of the blade's place (of its coordinates' bits): a sine hash loses precision
// far from the origin and smears into bands of alike blades
float hash3D(vec3 p, float seedOffset) {
    uvec3 q = floatBitsToUint(p);
    uint h = q.x * 374761393u + q.y * 668265263u + q.z * 3266489917u + floatBitsToUint(seedOffset) * 2246822519u;
    h = (h ^ (h >> 13u)) * 1274126177u;
    h ^= h >> 16u;
    h = (h ^ (h >> 15u)) * 2654435769u;
    h ^= h >> 16u;
    return float(h & 16777215u) / 16777216.0;
}

void main() {
    // 1. DYNAMIC ATLAS SELECTION
    // One of four blade shapes, and this blade's own width, lean, height and shade
    float atlasHash = hash3D(aInstancePos, float(worldSeed));
    float widthHash = hash3D(aInstancePos, float(worldSeed) + 17.0);
    float leanHash = hash3D(aInstancePos, float(worldSeed) + 31.0);
    float shadeHash = hash3D(aInstancePos, float(worldSeed) + 53.0);
    vec2 modifiedUV = aTexCoords;
    modifiedUV.x = (aTexCoords.x + floor(atlasHash * 4.0)) * 0.25;
    TexCoords = modifiedUV;
    fBladeShade = shadeHash;

    // 2. PASS-THROUGH PHENOTYPE FROM VBO STREAM
    fColourPhenotype = aColourPhenotype;
    float finalBladeHeight = aBladeHeight * (0.85 + 0.3 * fract(widthHash * 7.13));

    // 3. STABLE CYLINDRICAL BILLBOARDING & MESH TRANSFORM, about the planet's up where it stands
    vec3 localVertexPos = aPos;
    localVertexPos.y *= finalBladeHeight;
    localVertexPos.x *= 0.7 + 0.8 * widthHash;
    // Short grass is narrower too (full width from about 30 cm up), so a short blade is still a
    // blade standing up and not a wide strip lying along the ground
    // (the mesh is GRASS_BLADE_HEIGHT, 4.8 units, tall before its height scales it)
    localVertexPos.x *= clamp(finalBladeHeight * 4.8 / 3.0, 0.25, 1.0);
    // Blades lean a little either way, more at the tip
    localVertexPos.x += (leanHash - 0.5) * 0.35 * localVertexPos.y;

    vec3 base = aInstancePos - sphereCamera;
    vec3 billboardUp = normalize(aInstancePos);
    vec3 across = cross(billboardUp, base);
    vec3 billboardRight = dot(across, across) > 1e-8 ? normalize(across) : normalize(cross(billboardUp, vec3(0.0, 0.0, 1.0)));

    vec3 relative = base + (billboardRight * localVertexPos.x) + (billboardUp * localVertexPos.y);

    // 4. ZERO-POP WIND ANIMATION
    if (aPos.y > 0.1) {
        // (in waves passing across the land: neighbours sway nearly together)
        float sway = sin(time * 2.5 + dot(aInstancePos, vec3(0.85, 0.26, 0.46))) * 0.08 * finalBladeHeight;
        relative += billboardRight * sway;
    }

    fWorldPos = relative;
    gl_Position = viewProjection * vec4(relative, 1.0);
}
