#version 330 core

layout (location = 0) in vec3 aPos;
layout (location = 1) in vec2 aTexCoords;
layout (location = 2) in vec3 aInstancePos; 

out vec2 TexCoords;
out vec3 fWorldPos;

uniform mat4 view;
uniform mat4 projection;
uniform vec3 cameraPos; 
uniform float time;
uniform int worldSeed;

void main() {
    // 1. DYNAMIC ATLAS SELECTION
    float hash = fract(sin(dot(aInstancePos.xz, vec2(12.9898, 78.233))) * 43758.5453123);
    float textureSelect = step(0.5, hash); 

    vec2 modifiedUV = aTexCoords;
    modifiedUV.x = (aTexCoords.x * 0.5) + (textureSelect * 0.5);
    TexCoords = modifiedUV;

    // 2. STABLE CYLINDRICAL BILLBOARDING
    vec3 dirToCamera = cameraPos - aInstancePos;
    dirToCamera.y = 0.0; 
    dirToCamera = normalize(dirToCamera);

    vec3 billboardRight = vec3(-dirToCamera.z, 0.0, dirToCamera.x);
    vec3 billboardUp    = vec3(0.0, 1.0, 0.0); 

    vec3 worldPosition = aInstancePos 
                       + billboardRight * aPos.x 
                       + billboardUp * aPos.y;

    // 3. ZERO-POP WIND ANIMATION
    if (aPos.y > 0.1) { 
        float spatialOffset = sin(dot(aInstancePos.xz, vec2(12.9898, 78.233))) * float(worldSeed);
        float sway = sin(time * 2.5 + spatialOffset) * 0.15;
        worldPosition += billboardRight * sway; 
    }

    fWorldPos = worldPosition; 
    gl_Position = projection * view * vec4(worldPosition, 1.0);
}
