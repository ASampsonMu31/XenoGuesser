#version 330 core

// Per-vertex data (the basic flat quad mesh)
layout (location = 0) in vec3 aPos;
layout (location = 1) in vec2 aTexCoords;

// Per-instance data (passed via grassChunkCoordVBO)
layout (location = 2) in vec3 aInstancePos; 

out vec2 TexCoords;

uniform mat4 view;
uniform mat4 projection;
uniform float time;
uniform int worldSeed;

void main() {
    // 1. DYNAMIC ATLAS SELECTION (Fixes the paired grass issue)
    // Use GLSL's fract() to generate a deterministic hash between 0.0 and 1.0 based on instance positions
    float hash = fract(sin(dot(aInstancePos.xz, vec2(12.9898, 78.233))) * 43758.5453123);
    float textureSelect = step(0.5, hash); // Returns 0.0 (left blade) or 1.0 (right blade)

    // Compress incoming U coordinate by 50% and shift it right by 0.5 if textureSelect is 1.0
    vec2 modifiedUV = aTexCoords;
    modifiedUV.x = (aTexCoords.x * 0.5) + (textureSelect * 0.5);
    TexCoords = modifiedUV;

    // 2. CYLINDRICAL BILLBOARDING MATH
    vec3 billboardRight = vec3(view[0][0], view[1][0], view[2][0]);
    vec3 billboardUp    = vec3(0.0, 1.0, 0.0); // Keep grass perfectly upright

    // Calculate vertex world position relative to the base instance position
    vec3 worldPosition = aInstancePos 
                       + billboardRight * aPos.x 
                       + billboardUp * aPos.y;

    // 3. WIND ANIMATION
    if (aPos.y > 0.1) { // Only sway the top of the grass blade
        float uniqueOffset = sin(float(gl_InstanceID) * 0.05 + float(worldSeed));
        float sway = sin(time * 2.5 + uniqueOffset) * 0.15;
        worldPosition += billboardRight * sway; 
    }

    // 4. FINAL CLIP-SPACE TRANSFORMATION
    gl_Position = projection * view * vec4(worldPosition, 1.0);
}
