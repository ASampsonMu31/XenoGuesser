#version 330 core

layout (location = 0) in vec3 aPos;
layout (location = 1) in vec2 aTexCoords;
layout (location = 2) in vec3 aInstancePos; 
layout (location = 3) in float aColourPhenotype; // Driven directly from CPU
layout (location = 4) in float aBladeHeight;     // Driven directly from CPU

out vec2 TexCoords;
out vec3 fWorldPos;
out float fColourPhenotype;
out float fBladeShade; 

uniform mat4 view;
uniform mat4 projection;
uniform vec3 cameraPos; 
uniform float time;
uniform int worldSeed;
uniform float planetWidth;

float hash2D(vec2 p, float seedOffset) {
    return fract(sin(dot(p + seedOffset, vec2(127.1, 311.7))) * 43758.5453123);
}

// The planet's curve (see Planet): lowered by curvature times the square of the distance
// across the ground from curveCentre, along curveDown, the projection of a step up
uniform vec2 curveCentre;
uniform float curvature;
uniform vec4 curveDown;

vec4 planetDrop(vec3 world) {
    vec2 d = world.xz - curveCentre;
    return curveDown * (curvature * dot(d, d));
}

void main() {
    // 1. DYNAMIC ATLAS SELECTION
        // One of four blade shapes, and this blade's own width, lean, height and shade
    // Where the blade is on the planet, the same whichever side of the map's join it's seen from
    vec2 planetPos = vec2(mod(aInstancePos.x + planetWidth * 0.5, planetWidth), aInstancePos.z);
    float atlasHash = hash2D(planetPos, float(worldSeed));
    float widthHash = hash2D(planetPos, float(worldSeed) + 17.0);
    float leanHash = hash2D(planetPos, float(worldSeed) + 31.0);
    float shadeHash = hash2D(planetPos, float(worldSeed) + 53.0);
    vec2 modifiedUV = aTexCoords;
    modifiedUV.x = (aTexCoords.x + floor(atlasHash * 4.0)) * 0.25;
    TexCoords = modifiedUV;
    fBladeShade = shadeHash;

    // 2. PASS-THROUGH PHENOTYPE FROM VBO STREAM
    fColourPhenotype = aColourPhenotype;
        float finalBladeHeight = aBladeHeight * (0.85 + 0.3 * fract(widthHash * 7.13));

    // 3. STABLE CYLINDRICAL BILLBOARDING & MESH TRANSFORM
    vec3 localVertexPos = aPos;
        localVertexPos.y *= finalBladeHeight;
    localVertexPos.x *= 0.7 + 0.8 * widthHash;
    // Blades lean a little either way, more at the tip
    localVertexPos.x += (leanHash - 0.5) * 0.35 * localVertexPos.y; 

    vec3 dirToCamera = normalize(vec3(cameraPos.x - aInstancePos.x, 0.0, cameraPos.z - aInstancePos.z));
    vec3 billboardRight = vec3(-dirToCamera.z, 0.0, dirToCamera.x);
    vec3 billboardUp    = vec3(0.0, 1.0, 0.0); 

    vec3 worldPosition = aInstancePos + (billboardRight * localVertexPos.x) + (billboardUp * localVertexPos.y);

    // 4. ZERO-POP WIND ANIMATION
    if (aPos.y > 0.1) { 
        float sway = sin(time * 2.5 + aInstancePos.x) * 0.08 * finalBladeHeight;
        worldPosition += billboardRight * sway; 
    }

    fWorldPos = worldPosition; 
    gl_Position = projection * view * vec4(worldPosition, 1.0) - planetDrop(worldPosition);
}
