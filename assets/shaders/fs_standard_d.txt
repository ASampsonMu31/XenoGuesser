#version 330 core

// What the distance fades into: an even haze, a greyed version of the sky's colour (greyer
// still under cloud), the same in every direction (see Shader.setFog)
uniform vec3 fogColour;
uniform float fogReady;
vec3 fogHaze(vec3 fallback) {
    return fogReady > 0.5 ? fogColour : fallback;
}

// Snow (see XenoGuesser_GLEventListener.bindSnowCover): the climate at sea level over the
// whole sphere (its temperature, and how much rain or snow falls), a cube map read in the
// place's direction. Snow lies where it's colder than the water's freezing point, the higher
// the colder (so on mountain tops), wherever enough falls; and while it's snowing, fresh snow
// settles in growing patches over everywhere near or below freezing. It settles on what faces
// away from the planet's centre. Places are planet-centred: anything drawn on its own (about
// the origin, not on the planet) has none.
uniform samplerCube snowCover;
uniform vec2 snowState;      // whether it's ready, fresh snow (0 to 1)
uniform vec4 snowClimate;    // the freezing point, degrees below it for snow to lie thickly, degrees colder per unit up, sea level
uniform vec3 snowColour;
uniform float snowCleared;   // how much of the snow has been cleared off this (roads: see InfrastructureObject)
uniform float snowToday;     // how much warmer (Celsius) today is than usual round the player
uniform float planetRadius;
// (an exact hash of whole-numbered cells: a sine hash loses precision far from the origin and
// smears into stripes)
float hash3d(ivec3 c) {
    uint h = uint(c.x) * 374761393u + uint(c.y) * 668265263u + uint(c.z) * 2246822519u;
    h = (h ^ (h >> 13u)) * 1274126177u;
    h ^= h >> 16u;
    return float(h & 16777215u) / 16777216.0;
}
float noise3d(vec3 p) {
    vec3 f = fract(p);
    ivec3 i = ivec3(floor(p));
    f = f * f * (3.0 - 2.0 * f);
    float a = mix(mix(hash3d(i), hash3d(i + ivec3(1, 0, 0)), f.x), mix(hash3d(i + ivec3(0, 1, 0)), hash3d(i + ivec3(1, 1, 0)), f.x), f.y);
    float b = mix(mix(hash3d(i + ivec3(0, 0, 1)), hash3d(i + ivec3(1, 0, 1)), f.x), mix(hash3d(i + ivec3(0, 1, 1)), hash3d(i + ivec3(1, 1, 1)), f.x), f.y);
    return mix(a, b, f.z);
}
// Smooth noise along a line
float noise1d(float x) {
    float i = floor(x), f = x - i;
    f = f * f * (3.0 - 2.0 * f);
    return mix(hash3d(ivec3(int(i), 17, 29)), hash3d(ivec3(int(i) + 1, 17, 29)), f);
}
// How much of the surface snow covers where it faces a given way, before it's laid down as flakes
float snowLying(vec3 world, float facing) {
    float r = length(world);
    if (snowState.x < 0.5 || r < planetRadius * 0.5) return 0.0;
    vec2 climate = texture(snowCover, world / r).rg;
    float celsius = climate.r * 100.0 - 60.0 - max(0.0, r - planetRadius - snowClimate.w) * snowClimate.z;
    float falls = clamp((climate.g - 0.05) / 0.3, 0.0, 1.0);
    float lying = smoothstep(0.0, 1.0, (snowClimate.x - celsius) / snowClimate.y) * falls;
    // Fresh snow: flakes dotted about, then patches, then a full covering as it piles up
    float snowy = clamp((snowClimate.x + 3.0 - celsius - snowToday) / 3.0, 0.0, 1.0);
    // (never quite all: there are always gaps where the surface shows through)
    // (none under the sea or where the waves wash up: from the water's edge to a little
    // above the highest waves' reach)
    float washed = smoothstep(snowClimate.w + 8.0, snowClimate.w + 18.0, r - planetRadius);
    return min(0.8, max(lying, snowState.y * snowy)) * facing * (1.0 - snowCleared) * washed;
}
// Lays an amount of snow down as flakes and clumps over the surface as it is (fine dots, then
// patches, then drifts), each either snow or not, so a light dusting is scattered flakes and a
// heavy fall covers all but a few gaps
float snowFlakes(vec3 world, float amount) {
    if (amount <= 0.0) return 0.0;
    float speckle = noise3d(world * 5.3) * 0.38 + noise3d(world * 1.1) * 0.34 + noise3d(world * 0.19) * 0.28;
    // (spread out evenly, as the sum bunches round the middle, so the amount is the share covered)
    speckle = smoothstep(0.22, 0.78, speckle);
    return smoothstep(1.0 - amount, 1.0 - amount + 0.04, speckle);
}
float snowAt(vec3 world, vec3 n) {
    // Less on steep faces
    vec3 up = normalize(world);
    return snowFlakes(world, snowLying(world, smoothstep(0.45, 0.8, dot(n, up))));
}


in vec3 aPos;
in vec3 aWorldPos;   // planet-centred (aPos: the same, relative to the eyes)
in vec3 aNormal;
in vec2 aTexCoord;
in float PlanarDepth;

out vec4 fragColour;

struct Light {
    vec3 position;
    vec3 diffuse;
    vec3 specular;
};

struct Material {
    vec3 ambient;
    vec3 diffuse;
    vec3 specular;
    float shininess;
};

uniform sampler2D diffuse_texture;
uniform sampler2D skyTexture; 
uniform vec3 viewPos;

uniform int numLights;
uniform Light lights[64];

#include "shadows_include"

uniform Material material;
uniform vec2 scale; 
uniform vec3 ambientLight;
uniform mat4 skyRotation;

// Regional soil colour (terrain and tracks, on the sphere): colour ratios, stored quartered, in a
// cube map read in the place's direction
uniform samplerCube soilRegionMap;
uniform float useSoilRegions;
// Bare rock showing through wherever the ground is steep (terrain only)
uniform vec3 rockColour;
// Dirt tracks (above 0, how gritty): the same soil as the ground round them, a little worn,
// strewn with fine grit
uniform float trackGrit;

const float PI = 3.14159265359;

// A hash of a whole-numbered cell, exact however far from the origin (a sine hash loses
// precision out there and smears into stripes)
float cellHash(ivec2 c) {
    uvec2 q = uvec2(c) * uvec2(1664525u, 22695477u) + uvec2(1013904223u, 1u);
    uint h = q.x ^ (q.y * 747796405u);
    h ^= h >> 16u;
    h *= 2246822519u;
    h ^= h >> 13u;
    return float(h) / 4294967295.0;
}

float cellNoise(vec2 p) {
    vec2 f = fract(p);
    ivec2 i = ivec2(floor(p));
    f = f * f * (3.0 - 2.0 * f);
    return mix(mix(cellHash(i), cellHash(i + ivec2(1, 0)), f.x), mix(cellHash(i + ivec2(0, 1)), cellHash(i + ivec2(1, 1)), f.x), f.y);
}

// Poor houses' walls and roofs (0 to 1): faded, streaked dark by rain, blotched with grime
// and with pale patches of lichen and flaking
uniform float weathering;

// Two coordinates along the ground at a planet-centred place, in world units: the two world
// axes furthest from the local up (they switch where two axes are equally near it, a line
// rarely under anything small enough to notice)
vec2 alongGround(vec3 world) {
    vec3 a = abs(world);
    if (a.y >= a.x && a.y >= a.z) return world.xz;
    if (a.x >= a.z) return world.zy;
    return world.xy;
}

vec3 weather(vec3 c, vec3 world) {
    if (weathering <= 0.0) return c;
    // (x and z along the ground, y the height: as on a wall standing on flat ground)
    vec2 g = alongGround(world);
    vec3 p = vec3(g.x, length(world) - planetRadius, g.y);
    float w = weathering;
    c = mix(c, vec3(dot(c, vec3(0.3, 0.59, 0.11))), 0.45 * w);
    float across = cellNoise(vec2(p.x * 0.45, p.y * 0.03)) * cellNoise(vec2(p.z * 0.45 + 31.0, p.y * 0.03 + 17.0));
    c *= 1.0 - 0.55 * w * smoothstep(0.18, 0.5, across);
    float grime = smoothstep(0.45, 0.75, cellNoise(p.xz * 0.07 + vec2(p.y * 0.09, 0.0)) * 0.65
            + cellNoise(p.xz * 0.33 - vec2(0.0, p.y * 0.3)) * 0.35);
    c = mix(c, vec3(0.16, 0.14, 0.1), 0.6 * w * grime);
    float lichen = smoothstep(0.62, 0.8, cellNoise(p.xz * 0.21 + vec2(p.y * 0.17, 3.0)));
    return mix(c, vec3(0.62, 0.66, 0.5), 0.55 * w * lichen);
}

void main() {
    vec3 materialPos = aWorldPos;
    vec2 scaledUV = aTexCoord * scale; 

    // Sample the base texture color
    vec3 sampledColor = texture(diffuse_texture, scaledUV).rgb;

    vec3 texColor = sampledColor;
    float rockAmount = 0.0, rockBump = 0.0, cliffSnow = 0.0;
    if (useSoilRegions > 0.5) {
        texColor *= texture(soilRegionMap, normalize(aWorldPos)).rgb * 4.0;

        // Sand along the shore: from the water's edge to a little above the waves' reach (the
        // grass begins there, see BEACH_GRASS_FROM), wandering up and down a little, on the
        // gentler ground (steep shores stay rock, below), taking a little of the soil's tint
        if (snowState.x > 0.5) {
            float aboveSea = length(aWorldPos) - planetRadius - snowClimate.w;
            float level = dot(normalize(aNormal), normalize(aWorldPos));
            float beach = (1.0 - smoothstep(12.0, 20.0, aboveSea + (noise3d(aWorldPos * 0.03) - 0.5) * 6.0)) * smoothstep(0.84, 0.94, level);
            vec3 sand = mix(vec3(0.86, 0.79, 0.62), texColor * 1.6, 0.18) * (0.9 + 0.2 * noise3d(aWorldPos * 0.7));
            texColor = mix(texColor, sand, beach);
        }

        // Steep slopes are bare cliff: the same stone as the loose rocks, rough and speckled,
        // with fine bedding lines running across it (by height above the datum, wandering only a
        // little) and a few cracks. 3D noise of the place, exact however far it is from the
        // planet's centre, and the same whichever way the ground faces
        vec3 surfaceUp = normalize(aWorldPos);
        float steep = 1.0 - dot(normalize(aNormal), surfaceUp);
        float patchy = noise3d(materialPos * 0.05);
        rockAmount = smoothstep(0.14, 0.32, steep + (patchy - 0.5) * 0.12);
        // How far the height changes across a pixel (taken here, outside the branch below, where
        // the neighbouring pixels' values are all there; from the eye-relative place, which
        // keeps its precision)
        float heightStep = length(vec2(dot(dFdx(aPos), surfaceUp), dot(dFdy(aPos), surfaceUp)));
        if (rockAmount > 0.0) {
            float near = 1.0 - smoothstep(150.0, 400.0, length(viewPos - aPos));
            float height = length(aWorldPos) - planetRadius;
            // Bedding: thin lines, some beds thin and some thick, some lines fainter or missing,
            // faded out before they're too fine to draw. They rise and fall across the cliff by
            // an amount that depends only on where it is (the place straight down on the datum),
            // so every line at a place is lifted alike and they run on together: broad tilts of
            // several metres, folds, and jagged kinks (ridged noise, sharp where it turns)
            vec3 under = surfaceUp * planetRadius;
            float dip = (noise3d(under * 0.006) - 0.5) * 60.0
                    + (noise3d(under * 0.025 + 31.0) - 0.5) * 20.0
                    + abs(noise3d(under * 0.09 + 57.0) - 0.5) * 12.0
                    + abs(noise3d(under * 0.3 + 83.0) - 0.5) * 3.0;
            // (the beds' thicknesses: the height stretched and squeezed, the same everywhere at a
            // height, so lines bunch together in places and spread apart in others)
            float t = height + dip;
            float bedded = t + (noise1d(t * 0.07) - 0.5) * 9.0 + (noise1d(t * 0.31 + 13.0) - 0.5) * 2.0;
            float layer = bedded * 0.42;
            float across = abs(fract(layer) - 0.5) * 2.0;   // 1 on a line, 0 between
            // (how far the lines move across a pixel: the height's step, more where they slope)
            float layerStep = heightStep * 0.42 * 2.5;
            float lineWidth = max(0.12, layerStep * 2.0);
            float lineFade = 1.0 - smoothstep(0.25, 0.6, layerStep);
            float line = smoothstep(1.0 - lineWidth, 1.0, across) * lineFade;
            float layerHash = hash3d(ivec3(int(floor(layer)), 7, 3));
            line *= smoothstep(0.15, 0.5, layerHash) * (0.6 + 0.4 * noise3d(materialPos * 0.04 + 9.0));
            // Grain as on the loose rocks, at the cliff's larger scale too
            float grain = 0.84 + 0.2 * noise3d(materialPos * 0.9) + 0.12 * noise3d(materialPos * 4.3) * near;
            float blotch = noise3d(materialPos * 0.13 + 3.0);
            float crack = smoothstep(0.035, 0.0, abs(noise3d(materialPos * 0.1 + 5.0) - 0.5)) * (0.4 + 0.6 * near);
            vec3 tint = rockColour * (0.9 + 0.2 * noise3d(materialPos * 0.002)) * (0.92 + 0.12 * layerHash);
            vec3 rock = tint * grain * (0.88 + 0.24 * blotch) * (1.0 - 0.22 * line) * (1.0 - 0.35 * crack);
            texColor = mix(texColor, rock, rockAmount);
            // Rough to the light: knobbly, with the bedding standing out as small ledges
            rockBump = rockAmount * near * (noise3d(materialPos * 0.45) * 0.5 + noise3d(materialPos * 1.6) * 0.25 - line * 0.15);
            // Snow catches on the ledges and in hollows even where it's steep
            float ledge = smoothstep(0.55, 0.9, fract(layer + 0.12)) * smoothstep(0.3, 0.6, noise3d(materialPos * 0.3 + 11.0));
            float hollow = smoothstep(0.62, 0.8, noise3d(materialPos * 0.08 + 21.0));
            cliffSnow = rockAmount * snowFlakes(materialPos, snowLying(materialPos, smoothstep(0.15, 0.6, dot(normalize(aNormal), surfaceUp)))
                    * max(ledge, hollow) * 1.2);
        }
    }

    if (trackGrit > 0.0) {
        // Worn a little paler than the ground beside it, in soft patches
        vec2 gritPosition = alongGround(aWorldPos);
        texColor *= 1.08 + 0.07 * cellNoise(gritPosition * 0.35);
        // Grit: small stones of their own sizes and lumpy shapes, each sitting somewhere in
        // its own unit cell (more cells holding one the grittier the track), lit a
        // little on one side; smoothed at their edges, and fading out with distance before
        // they can shimmer
        float near = 1.0 - smoothstep(60.0, 150.0, length(viewPos - aPos));
        if (near > 0.0) {
            vec2 gp = gritPosition;
            ivec2 base = ivec2(floor(gp));
            float pixel = length(fwidth(gp));
            float cover = 0.0, tone = 1.0;
            for (int dy = -1; dy <= 1; dy++) {
                for (int dx = -1; dx <= 1; dx++) {
                    ivec2 c = base + ivec2(dx, dy);
                    if (cellHash(c + ivec2(5, 9)) > 0.15 + 0.6 * trackGrit) continue;
                    vec2 d = gp - (vec2(c) + vec2(0.15) + 0.7 * vec2(cellHash(c), cellHash(c + ivec2(13, 7))));
                    float r = 0.14 + 0.22 * cellHash(c + ivec2(3, 21));
                    float angle = atan(d.y, d.x);
                    float lumpy = 1.0 + 0.22 * sin(angle * 3.0 + 6.283 * cellHash(c + ivec2(8, 2)))
                                      + 0.1 * sin(angle * 5.0 + 6.283 * cellHash(c + ivec2(1, 19)));
                    float dist = length(d) / (r * lumpy);
                    // (a pixel's width from the smoothly varying position, not the distance, which
                    // jumps where neighbouring cells meet and would draw lines along them)
                    float edge = max(pixel / (r * lumpy), 0.05);
                    float here = 1.0 - smoothstep(1.0 - edge, 1.0 + edge, dist);
                    if (here > cover) {
                        cover = here;
                        // Its own shade, lighter on the side facing the light and darker away
                        tone = 0.45 + 0.85 * cellHash(c + ivec2(11, 4)) - 0.25 * (d.x + d.y) / r;
                    }
                }
            }
            texColor *= mix(1.0, tone, cover * near);
        }
    }

    if (weathering > 0.0) {
        texColor = weather(texColor, materialPos);
    }

    // Snow lying on the ground, roofs and anything else facing up
    // (snow is brighter than the ground it covers, so it reads as snow once lit)
    float snow = max(snowAt(aWorldPos, normalize(aNormal)), cliffSnow);
    texColor = mix(texColor, min(snowColour * 1.4, vec3(1.0)), snow);

    // Multiplicative blending for ambient lighting
    vec3 ambient = ambientLight * material.ambient * texColor;
    
    vec3 norm = normalize(aNormal);
    if (useSoilRegions > 0.5) {
        // The cliff's roughness tilts the normal by the bump's slope across the screen (taken
        // here, outside the branch above, as with heightStep); flat under snow
        vec3 dpdx = dFdx(aPos), dpdy = dFdy(aPos);
        float bx = dFdx(rockBump), by = dFdy(rockBump);
        vec3 r1 = cross(dpdy, norm), r2 = cross(norm, dpdx);
        float det = dot(dpdx, r1);
        vec3 grad = sign(det) * (bx * r1 + by * r2) * (1.0 - snow);
        if (abs(det) > 1e-8) norm = normalize(abs(det) * norm - grad);
    }
    vec3 viewDir = normalize(viewPos - aPos);

    vec3 totalDiffuse = vec3(0.0);
    vec3 totalSpecular = vec3(0.0);

    for (int i = 0; i < numLights; i++) {
        Light light = lights[i];
        vec3 lightDir = normalize(light.position - aPos);
        // (the sun's light, less where something stands between: see shadows_include)
        float shade = i == 0 ? sunShadow(aPos, norm) : 1.0;  
        float diff = max(dot(norm, lightDir), 0.0) * shade;
        
        totalDiffuse += light.diffuse * material.diffuse * diff * texColor;
        
        vec3 halfwayDir = normalize(lightDir + viewDir);  
        float spec = pow(max(dot(norm, halfwayDir), 0.0), material.shininess) * shade;
        totalSpecular += light.specular * spec * material.specular;
    }

    vec3 result = ambient + totalDiffuse + totalSpecular;

    // --- SEAMLESS INSIDE-SPHERE FOG COORD MAPPING ---
    float radialDistance = length(viewPos - aPos);
    // (the sky's own colour is only the haze until the fog's colour is set: not worked out otherwise)
    vec3 dynamicSkyColour = fogColour;
    if (fogReady < 0.5) {
        vec3 worldLookDir = normalize(aPos - viewPos);
        mat3 invSkyRot = transpose(mat3(skyRotation));
        vec3 localSkyDir = invSkyRot * worldLookDir;
    
        float theta = atan(localSkyDir.x, localSkyDir.z);
        if (theta < 0.0) {
            theta += 2.0 * PI;
        }
        float u = theta / (2.0 * PI);
        float v = asin(clamp(localSkyDir.y, -1.0, 1.0)) / PI + 0.5;
    
        dynamicSkyColour = texture(skyTexture, vec2(u, v)).rgb;
    }

    // --- CUBIC DISTANCE BLENDING ---
    float fogStart = 1000.0;
    float fogEnd = 2200.0; 

    float fogFactor = (fogEnd - radialDistance) / (fogEnd - fogStart);
    fogFactor = clamp(fogFactor, 0.0, 1.0); 
    fogFactor = smoothstep(0.0, 1.0, fogFactor);

    vec3 finalColourWithFog = mix(fogHaze(dynamicSkyColour), result, fogFactor);

    fragColour = vec4(finalColourWithFog, 1.0);
}
