#version 330 core

// Inputs from vertex shader
in vec3 aPos; // Note: Ensure these match what your vertex shader actually outputs
in vec3 aNormal;
in vec2 aTexCoord;

out vec4 fragColour;

// Uniforms
uniform sampler2D diffuse_texture;
uniform vec3 viewPos;
uniform vec3 ambientLight;

// New Uniforms for regional leaf colors
uniform vec3 u_OuterLeafColor;
uniform vec3 u_InnerLeafColor;

void main() {
    // 1. Sample the texture (Source should use Pure Red for outer, Pure Blue for inner)
    vec4 texColor = texture(diffuse_texture, aTexCoord);
    
    // 2. Alpha Test: Discard transparent fragments immediately
    if (texColor.a < 0.5) {
        discard;
    }

    // 3. Color Replacement using Channel Masking
    // We isolate the red and blue channels to use as weights
    vec3 finalLeafColor = vec3(0.0);
    
    // If the texture has red, tint it with the outer color
    finalLeafColor += texColor.r * u_OuterLeafColor;
    
    // If the texture has blue, tint it with the inner color
    finalLeafColor += texColor.b * u_InnerLeafColor;
    
    // If there are parts of the texture that aren't red or blue (like a brown branch), 
    // you can keep the original color by calculating the remaining texture parts:
    float maskSum = clamp(texColor.r + texColor.b, 0.0, 1.0);
    finalLeafColor += (1.0 - maskSum) * texColor.rgb;

    // 4. Apply Lighting to your newly calculated color
    vec3 norm = normalize(aNormal);
    vec3 lightDir = normalize(vec3(0.5, 1.0, 0.3)); 
    float diff = max(dot(norm, lightDir), 0.2);
    
    vec3 result = (ambientLight + diff) * finalLeafColor;
    
    fragColour = vec4(result, 1.0);
}
