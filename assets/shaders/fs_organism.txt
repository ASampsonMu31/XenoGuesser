#version 330 core

// What the distance fades into: an even haze, a greyed version of the sky's colour (greyer
// still under cloud), the same in every direction (see Shader.setFog)
uniform vec3 fogColour;
uniform float fogReady;
vec3 fogHaze(vec3 fallback) {
    return fogReady > 0.5 ? fogColour : fallback;
}

// Organisms: a glossy carapace coloured by part, with the species' pattern on its body

in vec3 aPos;
in vec3 aNormal;
in vec2 vUV;
flat in int vPart;

out vec4 fragColour;

uniform vec3 viewPos;
uniform vec3 sunPos;
uniform vec3 sunColour;
uniform vec3 ambientLight;
uniform sampler2D skyTexture;
uniform mat4 skyRotation;

uniform vec3 baseColour;
uniform vec3 accentColour;
uniform vec3 bellyColour;
uniform vec3 limbColour;
uniform vec3 eyeColour;
uniform int patternType;     // 0 bands, 1 spots, 2 stripes along the body, 3 mottling, 4 plain, 5 zigzags, 6 checks, 7 diagonals
uniform float patternScale;
uniform float gloss;
// Above 0, drawn as glass this see-through: the sky reflected in it, more so at a glancing angle,
// and anything printed on it (dark on its texture) solid
uniform float glassAlpha;
uniform vec3 trimColour;     // boots, gloves and fittings
uniform float lampGlow;      // above 0: the eyes are a lamp, shining in eyeColour whatever the light
// Anything below the water line is seen through the water
uniform float waterLevel;
uniform vec3 waterTint;
// Optional cloth texture (the player's spacesuit); 0 leaves surfaces plain
uniform sampler2D detailTexture;
uniform float detailAmount;
// People: their hair, and the flag some wear on the chest of their top
uniform vec3 hairColour;
uniform vec3 bagColour;
uniform sampler2D flagTexture;
uniform int flagOnTop;
// Goods (produce and packets): their nation's packaging texture, which u and v point into
uniform sampler2D productTexture;

const float PI = 3.14159265359;

float hash(vec2 p) {
    return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453);
}

float valueNoise(vec2 p) {
    vec2 i = floor(p), f = fract(p);
    f = f * f * (3.0 - 2.0 * f);
    return mix(mix(hash(i), hash(i + vec2(1, 0)), f.x), mix(hash(i + vec2(0, 1)), hash(i + vec2(1, 1)), f.x), f.y);
}

vec3 bodyColour() {
    // u runs around the segment with its top at 0.25, v along it
    float dorsal = sin(vUV.x * 2.0 * PI);
    vec3 colour = mix(bellyColour, baseColour, smoothstep(-0.45, 0.1, dorsal));
    float mark = 0.0;
    if (patternType == 0) {
        mark = smoothstep(0.55, 0.62, sin(vUV.y * patternScale * PI) * 0.5 + 0.5);
    } else if (patternType == 1) {
        vec2 cell = vec2(vUV.x * patternScale * 2.0, vUV.y * patternScale);
        vec2 id = floor(cell);
        vec2 centre = id + 0.25 + 0.5 * vec2(hash(id), hash(id + 7.3));
        mark = 1.0 - smoothstep(0.18, 0.26, length(cell - centre));
    } else if (patternType == 2) {
        mark = smoothstep(0.6, 0.7, sin(vUV.x * 2.0 * PI * floor(patternScale)) * 0.5 + 0.5);
    } else if (patternType == 3) {
        mark = smoothstep(0.55, 0.7, valueNoise(vec2(vUV.x * 8.0, vUV.y * patternScale * 1.5)));
    } else if (patternType == 5) {
        // Zigzags running round
        float zig = abs(fract(vUV.x * 6.0) - 0.5) * 2.0;
        mark = smoothstep(0.35, 0.45, abs(fract(vUV.y * patternScale + zig * 0.35) - 0.5));
    } else if (patternType == 6) {
        // Checks
        vec2 cell = floor(vec2(vUV.x * floor(patternScale * 2.0 + 2.0), vUV.y * patternScale));
        mark = mod(cell.x + cell.y, 2.0);
    } else if (patternType == 7) {
        // Diagonal stripes
        mark = smoothstep(0.45, 0.55, fract(vUV.x * 5.0 + vUV.y * patternScale) );
    }
    // Markings show on the back, not the belly
    return mix(colour, accentColour, mark * smoothstep(-0.2, 0.3, dorsal));
}

void main() {
    vec3 albedo;
    float shine = gloss;
    if (vPart == 0) {
        albedo = bodyColour();
    } else if (vPart == 17) {
        // A top, perhaps with the flag on its chest (u runs round the body, the chest at 0.25)
        albedo = bodyColour();
        if (flagOnTop == 1) {
            vec2 f = vec2((vUV.x - 0.16) / 0.18, (vUV.y - 0.48) / 0.26);
            if (f.x > 0.0 && f.x < 1.0 && f.y > 0.0 && f.y < 1.0) albedo = texture(flagTexture, f).rgb;   // stored bottom row first, as v runs up the body
        }
    } else if (vPart == 18) {
        // A shopping bag: plastic or cloth, a little creased
        albedo = bagColour * (0.85 + 0.2 * valueNoise(vUV * vec2(6.0, 4.0)));
        shine = 0.5;
    } else if (vPart == 15) {
        albedo = texture(productTexture, vUV).rgb;
        shine = 0.45;
    } else if (vPart == 16) {
        // Hair, streaked along its strands
        float streak = valueNoise(vec2(vUV.x * 40.0, vUV.y * 3.0));
        albedo = hairColour * (0.75 + 0.45 * streak);
        shine = 0.35;
    } else if (vPart == 5) {
        albedo = eyeColour;
        shine = 1.0;
    } else if (vPart == 8) {
        // Bare skin
        albedo = limbColour;
        shine = 0.25;
    } else if (vPart == 7) {
        albedo = trimColour;
        shine = 0.5;
    } else if (vPart == 9) {
        // Polished brass
        albedo = vec3(0.80, 0.60, 0.26);
        shine = 0.9;
    } else if (vPart == 10) {
        // The compass card: cream, with a ring of ticks, longer at the cardinal points
        float rim = clamp((1.0 - vUV.y) * 4.0, 0.0, 1.0);   // 1 at the edge, 0 in the middle
        float eighth = vUV.x * 8.0;
        float tick = 1.0 - smoothstep(0.04, 0.09, abs(fract(eighth + 0.5) - 0.5) * (1.0 + 2.0 * rim));
        bool cardinal = mod(floor(eighth + 0.5), 2.0) < 0.5;
        float inner = cardinal ? 0.55 : 0.75;
        float ring = smoothstep(0.86, 0.88, rim) * (1.0 - smoothstep(0.9, 0.92, rim));
        float mark = max(tick * step(inner, rim) * step(rim, 0.86), ring);
        albedo = mix(vec3(0.93, 0.9, 0.8), vec3(0.12, 0.11, 0.1), mark);
        shine = 0.2;
    } else if (vPart == 11) {
        albedo = vec3(0.85, 0.12, 0.1);
        shine = 0.6;
    } else if (vPart == 12) {
        albedo = vec3(0.9, 0.9, 0.92);
        shine = 0.6;
    } else if (vPart == 14) {
        albedo = vec3(0.015, 0.015, 0.018);
        shine = 0.0;
    } else if (vPart == 13) {
        albedo = vec3(0.62, 0.64, 0.68);
        shine = 0.55;
    } else if (vPart == 6) {
        // Horns and pincers darken towards their tips
        albedo = mix(limbColour, limbColour * 0.25, vUV.y);
    } else {
        // Limbs, with darker joints at each end
        float joint = smoothstep(0.75, 1.0, abs(vUV.y * 2.0 - 1.0));
        albedo = mix(limbColour, limbColour * 0.45, joint);
        shine *= 0.7;
    }

    if (detailAmount > 0.0 && (vPart == 0 || vPart == 1 || vPart == 17)) {
        float cloth = texture(detailTexture, vUV * vec2(2.0, 3.0)).r;
        albedo *= mix(1.0, cloth * 1.08, detailAmount);
    }

    vec3 n = normalize(aNormal);
    vec3 viewDir = normalize(viewPos - aPos);
    if (dot(n, viewDir) < 0.0) n = -n;
    vec3 lightDir = normalize(sunPos - aPos);
    float diffuse = max(dot(n, lightDir), 0.0);
    vec3 halfway = normalize(lightDir + viewDir);
    float specular = pow(max(dot(n, halfway), 0.0), mix(12.0, 90.0, shine)) * shine * 0.8;
    // A faint rim of sky light picks out the silhouette
    float rim = pow(1.0 - max(dot(n, viewDir), 0.0), 3.0) * 0.25;

    vec3 colour = albedo * (ambientLight * 1.4 + sunColour * diffuse * 0.85) + sunColour * specular + rim * albedo;
    if (lampGlow > 0.0 && vPart == 5) colour = mix(colour, eyeColour, lampGlow);

    float submerged = waterLevel - aPos.y;
    if (submerged > 0.0) {
        colour = mix(colour, waterTint * (ambientLight * 1.4 + sunColour * 0.6), 0.25 + 0.35 * clamp(submerged / 10.0, 0.0, 1.0));
    }

    // Distance fog into the sky, as everywhere else
    float radialDistance = length(viewPos - aPos);
    vec3 localSkyDir = transpose(mat3(skyRotation)) * normalize(aPos - viewPos);
    float theta = atan(localSkyDir.x, localSkyDir.z);
    if (theta < 0.0) theta += 2.0 * PI;
    vec3 skyColour = texture(skyTexture, vec2(theta / (2.0 * PI), asin(clamp(localSkyDir.y, -1.0, 1.0)) / PI + 0.5)).rgb;
    float fogFactor = smoothstep(0.0, 1.0, clamp((2200.0 - radialDistance) / 1200.0, 0.0, 1.0));

    float alpha = 1.0;
    if (glassAlpha > 0.0) {
        vec3 bounced = transpose(mat3(skyRotation)) * reflect(-viewDir, n);
        float bt = atan(bounced.x, bounced.z);
        if (bt < 0.0) bt += 2.0 * PI;
        vec3 reflected = texture(skyTexture, vec2(bt / (2.0 * PI), asin(clamp(bounced.y, -1.0, 1.0)) / PI + 0.5)).rgb;
        float fresnel = pow(1.0 - max(dot(n, viewDir), 0.0), 3.0);
        float ink = smoothstep(0.82, 0.45, dot(albedo, vec3(0.3, 0.59, 0.11)));
        colour = mix(reflected * 0.9 + sunColour * specular * 2.0, albedo * (ambientLight * 1.4 + sunColour * 0.6), ink);
        alpha = clamp(max(glassAlpha + 0.55 * fresnel + specular, ink), 0.0, 1.0);
    }
    fragColour = vec4(mix(fogHaze(skyColour), colour, fogFactor), alpha);
}
