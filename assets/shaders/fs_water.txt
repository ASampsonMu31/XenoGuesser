#version 330 core

in vec3 aPos;
in vec3 aNormal;
in vec2 aTexCoord;
in float PlanarDepth;

out vec4 fragColour;

uniform vec3 viewPos;
uniform float seaLevelHeight;
uniform vec2 windowSize;

uniform sampler2D diffuse_texture;     
uniform sampler2D terrainDepthTexture; 
uniform sampler2D skyTexture; 

uniform vec3 sunPos;         
uniform vec3 lightSpecular;  
uniform vec3 matSpecular;    
uniform float matShininess;  

uniform float time;
uniform vec3 ambientLight;   
uniform mat4 skyRotation;

// Seeded per world: colour seen through shallow water, and of open deep ocean
uniform vec3 u_ShallowTint;
uniform vec3 u_DeepOceanColour;

const float SHALLOW_OPACITY_FLOOR = 0.09;  
const float EDGE_FADE_DISTANCE    = 0.5;   
const float DEPTH_FALLOFF_SPEED   = 0.20;  
const float RIPPLE_SPEED          = 0.04;
const float RIPPLE_STRENGTH       = 9.0;
const float PI                    = 3.14159265359;

float nearPlane = 10.0;
float farPlane = 2500.0;

float linearizeDepth(float rawDepth) {
    float zNDC = rawDepth * 2.0 - 1.0; 
    return (2.0 * nearPlane * farPlane) / (farPlane + nearPlane - zNDC * (farPlane - nearPlane));
}

void main() {
    vec3 viewDir = normalize(viewPos - aPos);
    vec2 screenUV = gl_FragCoord.xy / windowSize;
    
    float rawDepth = texture(terrainDepthTexture, screenUV).r;
    float landDistance = linearizeDepth(rawDepth);
    float waterDistance = PlanarDepth;
    
    float totalThickness = max(landDistance - waterDistance, 0.0);
    
    float cosTheta = dot(viewDir, vec3(0.0, 1.0, 0.0));
    cosTheta = max(abs(cosTheta), 0.0001);
    
    float actualRayPathLength = totalThickness / cosTheta;
    float stretchedPathLength = actualRayPathLength * DEPTH_FALLOFF_SPEED;

    vec3 absorptionCoefficients = vec3(0.015, 0.005, 0.001);
    vec3 transparencyFactor = exp(-stretchedPathLength * absorptionCoefficients);

    vec2 worldUV1 = (aPos.xz * 0.010) + vec2(0.0, time * RIPPLE_SPEED);
    vec2 worldUV2 = (aPos.xz * 0.011) + vec2(0.0, -time * RIPPLE_SPEED * 1.01);
    vec2 worldUV3 = (aPos.xz * 0.009) + vec2(time * RIPPLE_SPEED * 1.02, 0.0);
    vec2 worldUV4 = (aPos.xz * 0.010) + vec2(-time * RIPPLE_SPEED * 1.03, 0.0);
    
    vec3 tex1 = texture(diffuse_texture, worldUV1).rgb;
    vec3 tex2 = texture(diffuse_texture, worldUV2).rgb;
    vec3 tex3 = texture(diffuse_texture, worldUV3).rgb;
    vec3 tex4 = texture(diffuse_texture, worldUV4).rgb;

    vec3 combinedTex = (tex1 * tex2) + (tex3 * tex4);
    vec3 waterSurfaceTex = clamp(combinedTex * 1.5, 0.0, 1.0);

    vec3 shallowColourTint = u_ShallowTint;
    vec3 deepOceanColour = u_DeepOceanColour;
    
    float factor = transparencyFactor.g;
    vec3 shallowWaterBlend = mix(shallowColourTint, waterSurfaceTex, factor);
    vec3 waterBaseColour = mix(deepOceanColour, shallowWaterBlend, factor);

    float edgeFade = clamp(totalThickness / EDGE_FADE_DISTANCE, 0.0, 1.0);
    float baseOpacity = mix(SHALLOW_OPACITY_FLOOR, 1.0, 1.0 - factor);
    float opacity = baseOpacity * edgeFade;

        // Ripple normals: the slope of the blended surface pattern, worked out in world space
    // from screen-space derivatives, so the sun's reflection breaks up across the ripples
    float rippleHeight = dot(combinedTex, vec3(0.333));
    vec2 heightDerivs = vec2(dFdx(rippleHeight), dFdy(rippleHeight));
    mat2 worldDerivs = mat2(dFdx(aPos.xz), dFdy(aPos.xz));
    float det = determinant(worldDerivs);
    vec2 slope = abs(det) > 1e-6 ? inverse(transpose(worldDerivs)) * heightDerivs : vec2(0.0);
    vec3 normal = normalize(vec3(-slope.x * RIPPLE_STRENGTH, 1.0, -slope.y * RIPPLE_STRENGTH)); 
    vec3 lightDir = normalize(sunPos - aPos);
    vec3 halfwayDir = normalize(lightDir + viewDir);
    
    float facing = max(dot(normal, halfwayDir), 0.0);
    // A tight, bright core plus a broader glitter that the ripples scatter across the water
    vec3 specularHighlight = lightSpecular * (matSpecular * pow(facing, matShininess) + 0.9 * pow(facing, 160.0));

    vec3 baseWaterColourWithSpecular = waterBaseColour + specularHighlight;

    // --- PERFECT SKY-SPHERE INVERSION TRANSFORM ---
    float radialDistance = length(viewPos - aPos);
    vec3 worldLookDir = normalize(aPos - viewPos);
    mat3 invSkyRot = transpose(mat3(skyRotation));
    vec3 localSkyDir = invSkyRot * worldLookDir;
    
    float theta = atan(localSkyDir.x, localSkyDir.z);
    if (theta < 0.0) {
        theta += 2.0 * PI;
    }
    float u = theta / (2.0 * PI);
    float v = asin(clamp(localSkyDir.y, -1.0, 1.0)) / PI + 0.5;
    
    vec3 dynamicSkyColour = texture(skyTexture, vec2(u, v)).rgb;

    // --- CUBIC DISTANCE BLENDING ---
    float fogStart = 1000.0;
    float fogEnd = 2200.0; 

    float fogFactor = (fogEnd - radialDistance) / (fogEnd - fogStart);
    fogFactor = clamp(fogFactor, 0.0, 1.0); 
    fogFactor = smoothstep(0.0, 1.0, fogFactor);

    vec3 finalColourWithFog = mix(dynamicSkyColour, baseWaterColourWithSpecular, fogFactor);

    // --- CRITICAL ALPHA FIX FOR DISTANT WATER ---
    // As fogFactor goes to 0.0 (maximum fog at a distance), opacity must force-climb to 1.0
    // This prevents the sky-coloured fog pixels from rendering as transparent cutouts.
    float finalOpacity = mix(1.0, opacity, fogFactor);

    fragColour = vec4(finalColourWithFog, finalOpacity);
}
