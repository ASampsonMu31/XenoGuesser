#version 330 core
in vec2 TexCoords;
out vec4 fragColor;

uniform sampler2D grassTexture;
uniform vec3 ambientLight;
uniform vec3 sunDir; 
uniform vec3 sunColor;

void main() {
    vec4 texColor = texture(grassTexture, TexCoords);
    
    // Strict cut-off point for transparent PNG borders 
    if (texColor.a < 0.25) {
        discard;
    }
    
    // Upward normal vector for soft ground lighting distribution
    vec3 fakeNormal = vec3(0.0, 1.0, 0.0); 
    vec3 lightDir = normalize(sunDir);
    
    // Half-Lambert calculation to keep shaded grass visible on non-lit faces
    float diff = dot(fakeNormal, lightDir) * 0.5 + 0.5;
    vec3 diffuseLight = sunColor * diff;
    
    vec3 finalLight = ambientLight + diffuseLight;
    finalLight = clamp(finalLight, 0.0, 1.5);
    
    fragColor = vec4(texColor.rgb * finalLight, texColor.a);
}
