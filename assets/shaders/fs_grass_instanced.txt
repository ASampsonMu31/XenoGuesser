#version 330 core
in vec2 TexCoords;
out vec4 fragColor;

uniform sampler2D grassTexture;
uniform vec3 ambientLight;

void main() {
    vec4 texColor = texture(grassTexture, TexCoords);
    
    // 1. Drop completely transparent background pixels safely
    if (texColor.a < 0.05) {
        discard;
    }
    
    // 2. Purely Ambient Lighting Calculation
    // We clamp the light modifier to a reasonable range to prevent extreme darks/brights
    vec3 finalLight = clamp(3*ambientLight, 0.0, 1.5);
    
    // 3. Alpha-to-Coverage distance preservation math
    // Smooths out and stabilizes the fine texture details as they shrink away from the camera
    float adjustedAlpha = texColor.a;
    adjustedAlpha = (adjustedAlpha - 0.5) / max(fwidth(adjustedAlpha), 0.0001) + 0.5;
    adjustedAlpha = clamp(adjustedAlpha, 0.0, 1.0);
    
    // Final output combining texture color, flat ambient light, and filtered alpha
    fragColor = vec4(texColor.rgb * finalLight, adjustedAlpha);
}
