#version 330 core

in vec2 TexCoords;
in vec3 fWorldPos;
in float fColourPhenotype; 

out vec4 fragColour;

uniform sampler2D grassTexture;
uniform sampler2D skyTexture; 
uniform vec3 ambientLight;
uniform vec3 cameraPos;  
uniform mat4 skyRotation;

const float PI = 3.14159265359;

// Natural colour palette boundaries
const vec3 lightBrown = vec3(0.54, 0.43, 0.31);
const vec3 darkGreen  = vec3(0.08, 0.32, 0.10);

void main() {
    vec4 texColour = texture(grassTexture, TexCoords);
    
    if (texColour.a < 0.05) {
        discard;
    }
    
    // Applying a power curve skews variations toward 0.0 (dark forest green).
    // This creates lush green grass overall, broken up by natural patches of light brown.
    float biasedPhenotype = fColourPhenotype;
    
    // Mix equation colour blending
    vec3 proceduralGrassColour = mix(lightBrown, darkGreen, biasedPhenotype);
    
    vec3 finalLight = clamp(3.0 * ambientLight, 0.0, 1.5);
    vec3 baseGrassColour = proceduralGrassColour * finalLight;
    
    // Anti-Aliasing Alpha Edge Sharpening
    float adjustedAlpha = texColour.a;
    adjustedAlpha = (adjustedAlpha - 0.5) / max(fwidth(adjustedAlpha), 0.0001) + 0.5;
    adjustedAlpha = clamp(adjustedAlpha, 0.0, 1.0);
    
    // --- SKY-SPHERE INVERSION TRANSFORM ---
    float radialDistance = length(cameraPos - fWorldPos);
    vec3 worldLookDir = normalize(fWorldPos - cameraPos);
    mat3 invSkyRot = transpose(mat3(skyRotation));
    vec3 localSkyDir = invSkyRot * worldLookDir;
    
    float theta = atan(localSkyDir.x, localSkyDir.z);
    if (theta < 0.0) {
        theta += 2.0 * PI;
    }
    float u = theta / (2.0 * PI);
    float v = asin(clamp(localSkyDir.y, -1.0, 1.0)) / PI + 0.5;
    
    vec3 dynamicSkyColour = texture(skyTexture, vec2(u, v)).rgb;

    // --- CUBIC DISTANCE FOG BLENDING ---
    float fogStart = 1000.0;
    float fogEnd = 2200.0; 

    float fogFactor = (fogEnd - radialDistance) / (fogEnd - fogStart);
    fogFactor = clamp(fogFactor, 0.0, 1.0); 
    fogFactor = smoothstep(0.0, 1.0, fogFactor);

    vec3 finalColourWithFog = mix(dynamicSkyColour, baseGrassColour, fogFactor);

    fragColour = vec4(finalColourWithFog, adjustedAlpha);
}
