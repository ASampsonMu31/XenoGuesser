#version 330 core

// The cloud layer (see SphereWeather.java, whose cloud noise this matches exactly): a shell
// high above, fixed to the world and drifting with the wind, stretched along the wind and
// streaked across it so the clouds look fibrous. Its cover and darkness follow the rainfall.
// The noise is laid out on the day's wind circle: (u, v) = along it and off it, along the
// ground (axisA the up where the day began, axisB the wind's way there, axisW = axisA x axisB).

in vec2 uv;
out vec4 fragColour;

uniform vec3 cameraForward, cameraRight, cameraUp;
uniform float aspectRatio;
uniform vec3 eyeUp;          // straight up at the eyes
uniform vec3 eyePosition;    // the eyes, planet-centred
uniform float eyeRadius;     // their distance from the planet's centre
uniform float riseSquared;   // the layer's radius squared less eyeRadius squared
uniform float planetRadius;
uniform vec3 axisA, axisB, axisW;
uniform float scale, drift;
uniform float coverage, darkness, daylight;
uniform vec3 skyColour;
uniform vec2 dayOffset;
uniform float streak;       // how streaked along the wind today's clouds are (0 none, 1 most)
uniform vec3 sunDirection;  // towards the sun, in the world
uniform vec2 sunAlong;      // the sun's way in (u, v) at the eyes
uniform vec3 sunColour;
uniform float shading;
uniform vec4 haze;         // what the distance fades into: the clouds' usual colour, and how much
uniform float horizonDip;  // how far below level the sea's horizon is (the sine of its angle)

float hash(ivec2 c) {
    uint h = uint(c.x) * 374761393u + uint(c.y) * 668265263u;
    h = (h ^ (h >> 13u)) * 1274126177u;
    h = h ^ (h >> 16u);
    return float(h & 16777215u) / 16777216.0;
}

float noise(vec2 p) {
    vec2 i = floor(p);
    vec2 f = p - i;
    ivec2 c = ivec2(i);
    vec2 u = f * f * (3.0 - 2.0 * f);
    float a = hash(c), b = hash(c + ivec2(1, 0)), d0 = hash(c + ivec2(0, 1)), d1 = hash(c + ivec2(1, 1));
    float bottom = a + (b - a) * u.x, top = d0 + (d1 - d0) * u.x;
    return bottom + (top - bottom) * u.y;
}

float fbm(vec2 p, int octaves) {
    float value = 0.0, amplitude = 0.5, total = 0.0;
    for (int i = 0; i < octaves; i++) {
        value += amplitude * noise(p);
        total += amplitude;
        p = p * 2.03 + vec2(17.3, 9.1);
        amplitude *= 0.5;
    }
    return value / total;
}

// At cloud coordinates (u, v): along the wind (u) and across it (v)
float cloudNoise(vec2 place) {
    float a = place.x / scale + drift + dayOffset.x, b = place.y / scale + dayOffset.y;
    vec2 q = vec2(a * mix(1.0, 0.45, streak), b * mix(1.0, 1.6, streak));
    float w = fbm(q * 0.7 + vec2(3.1, 7.7), 3);
    float n = fbm(q + vec2(w - 0.5) * 2.2, 5);
    float f = fbm(vec2(a * 0.9, b * 6.0) + vec2(w * 1.5), 3);
    return n * (1.0 - 0.2 * streak) + f * 0.2 * streak;
}

void main() {
    vec2 ndc = uv * 2.0 - 1.0;
    vec3 ray = normalize(cameraForward + cameraRight * (ndc.x * aspectRatio * 0.41421356)
            + cameraUp * (ndc.y * 0.41421356));
    float rayUp = dot(ray, eyeUp);
    // Towards the horizon the layer is far off and seen edge on: it merges into the haze
    // that the distant land fades into too. (The horizon is the sea's, which on so small a
    // planet lies a few degrees below level: the haze starts there, not at eye level, else
    // a band of it shows below level, over the sea's far edge.)
    float horizon = smoothstep(-horizonDip, 0.22 - horizonDip, rayUp);
    vec4 cloud = vec4(0.0);
    if (rayUp > 0.005 - horizonDip) {
        // Where the ray meets the layer (in the form that keeps its precision looking up)
        float b = eyeRadius * rayUp;
        float t = riseSquared / (b + sqrt(b * b + riseSquared));
        vec3 hit = normalize(eyePosition + ray * t);
        vec2 place = vec2(atan(dot(hit, axisB), dot(hit, axisA)), asin(clamp(dot(hit, axisW), -1.0, 1.0))) * planetRadius;
        float n = cloudNoise(place);
        float threshold = 0.72 - 0.50 * coverage;
        float cover = smoothstep(threshold, threshold + 0.18, n);
        float thickness = smoothstep(threshold, threshold + 0.4, n);
        float shade = shading * thickness * (0.2 + 0.8 * darkness);
        // Lit white by day (dimmer and sky-tinted at dusk and night), grey where thick and wet
        vec3 lit = mix(skyColour * 0.35 + vec3(0.06), vec3(0.97, 0.97, 0.99), daylight);
        vec3 dark = lit * vec3(0.30, 0.31, 0.34);
        vec3 colour = mix(lit, dark, shade);
        // The sun lights each cloud on the side it shines from: where the cloud thins out
        // towards the sun, its edge is lit bright, dark storm clouds as much as any
        vec2 towardsSun = normalize(sunAlong + vec2(1e-4));
        float nearer = cloudNoise(place + towardsSun * scale * 0.06);
        float sunlitEdge = clamp((n - nearer) * 7.0, 0.0, 1.0);
        // and looking towards the sun, light shines through the clouds, thin ones most
        float through = pow(max(dot(ray, normalize(sunDirection)), 0.0), 6.0);
        float glow = (0.55 * sunlitEdge + 0.8 * through * (1.0 - 0.55 * thickness)) * daylight;
        colour = min(colour + sunColour * glow, vec3(1.0));
        cloud = vec4(colour, cover * 0.95);
    }
    vec4 result = mix(haze, cloud, horizon);
    if (result.a <= 0.002) discard;
    fragColour = result;
}
