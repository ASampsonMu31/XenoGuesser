#version 330 core

// What the distance fades into: an even haze, a greyed version of the sky's colour (greyer
// still under cloud), the same in every direction (see Shader.setFog)
uniform vec3 fogColour;
uniform float fogReady;
vec3 fogHaze(vec3 fallback) {
    return fogReady > 0.5 ? fogColour : fallback;
}

// Rocks: matte stone with fine grain, lit by the sun and sky, fading into the distance

in vec3 aPos;
in vec3 aGrain;
in vec3 aNormal;
in vec3 aColour;
in vec3 aWorld;
out vec4 fragColour;

// Snow (see XenoGuesser_GLEventListener.bindSnowCover): the climate at sea level over the
// whole sphere (its temperature, and how much rain or snow falls), a cube map read in the
// place's direction. Snow lies where it's colder than the water's freezing point, the higher
// the colder (so on mountain tops), wherever enough falls; and while it's snowing, fresh snow
// settles in growing patches over everywhere near or below freezing. It settles on what faces
// away from the planet's centre. Places are planet-centred: anything drawn on its own (about
// the origin, not on the planet) has none.
uniform samplerCube snowCover;
uniform vec2 snowState;      // whether it's ready, fresh snow (0 to 1)
uniform vec4 snowClimate;    // the freezing point, degrees below it for snow to lie thickly, degrees colder per unit up, sea level
uniform vec3 snowColour;
uniform float snowCleared;   // how much of the snow has been cleared off this (roads: see InfrastructureObject)
uniform float snowToday;     // how much warmer (Celsius) today is than usual round the player
uniform float planetRadius;
// (an exact hash of whole-numbered cells: a sine hash loses precision far from the origin and
// smears into stripes)
float hash3d(ivec3 c) {
    uint h = uint(c.x) * 374761393u + uint(c.y) * 668265263u + uint(c.z) * 2246822519u;
    h = (h ^ (h >> 13u)) * 1274126177u;
    h ^= h >> 16u;
    return float(h & 16777215u) / 16777216.0;
}
float noise3d(vec3 p) {
    vec3 f = fract(p);
    ivec3 i = ivec3(floor(p));
    f = f * f * (3.0 - 2.0 * f);
    float a = mix(mix(hash3d(i), hash3d(i + ivec3(1, 0, 0)), f.x), mix(hash3d(i + ivec3(0, 1, 0)), hash3d(i + ivec3(1, 1, 0)), f.x), f.y);
    float b = mix(mix(hash3d(i + ivec3(0, 0, 1)), hash3d(i + ivec3(1, 0, 1)), f.x), mix(hash3d(i + ivec3(0, 1, 1)), hash3d(i + ivec3(1, 1, 1)), f.x), f.y);
    return mix(a, b, f.z);
}
float snowAt(vec3 world, vec3 n) {
    float r = length(world);
    if (snowState.x < 0.5 || r < planetRadius * 0.5) return 0.0;
    vec3 up = world / r;
    vec2 climate = texture(snowCover, up).rg;
    float celsius = climate.r * 100.0 - 60.0 - max(0.0, r - planetRadius - snowClimate.w) * snowClimate.z;
    float falls = clamp((climate.g - 0.05) / 0.3, 0.0, 1.0);
    float lying = smoothstep(0.0, 1.0, (snowClimate.x - celsius) / snowClimate.y) * falls;
    // Fresh snow: flakes dotted about, then patches, then a full covering as it piles up
    float snowy = clamp((snowClimate.x + 3.0 - celsius - snowToday) / 3.0, 0.0, 1.0);
    // How much of the surface is under snow, less on steep faces...
    // (never quite all: there are always gaps where the surface shows through)
    // (none under the sea or where the waves wash up: from the water's edge to a little
    // above the highest waves' reach)
    float washed = smoothstep(snowClimate.w + 8.0, snowClimate.w + 18.0, r - planetRadius);
    float amount = min(0.8, max(lying, snowState.y * snowy)) * smoothstep(0.45, 0.8, dot(n, up)) * (1.0 - snowCleared) * washed;
    if (amount <= 0.0) return 0.0;
    // ...laid down as flakes and clumps of snow over the surface as it is (fine dots, then
    // patches, then drifts), each either snow or not, so a light dusting is scattered flakes
    // and a heavy fall covers all but a few gaps
    float speckle = noise3d(world * 5.3) * 0.38 + noise3d(world * 1.1) * 0.34 + noise3d(world * 0.19) * 0.28;
    // (spread out evenly, as the sum bunches round the middle, so the amount is the share covered)
    speckle = smoothstep(0.22, 0.78, speckle);
    return smoothstep(1.0 - amount, 1.0 - amount + 0.04, speckle);
}

uniform vec3 viewPos;
uniform vec3 sunPos;
uniform vec3 sunColour;
uniform vec3 ambientLight;
uniform sampler2D skyTexture;
uniform mat4 skyRotation;
// Straight up at the camera (the rocks drawn are near enough that it is up for them too)
uniform vec3 up;

const float PI = 3.14159265359;

float hash(vec3 p) {
    return fract(sin(dot(p, vec3(127.1, 311.7, 74.7))) * 43758.5453);
}

// Smooth value noise: blends between the lattice points instead of stepping, so no blocks
float valueNoise(vec3 p) {
    vec3 i = floor(p), f = fract(p);
    f = f * f * (3.0 - 2.0 * f);
    float a = mix(mix(hash(i), hash(i + vec3(1, 0, 0)), f.x), mix(hash(i + vec3(0, 1, 0)), hash(i + vec3(1, 1, 0)), f.x), f.y);
    float b = mix(mix(hash(i + vec3(0, 0, 1)), hash(i + vec3(1, 0, 1)), f.x), mix(hash(i + vec3(0, 1, 1)), hash(i + vec3(1, 1, 1)), f.x), f.y);
    return mix(a, b, f.z);
}

void main() {
    vec3 n = normalize(aNormal);
    vec3 lightDir = normalize(sunPos - aPos);
    float diffuse = max(dot(n, lightDir), 0.0);
    // Speckled grain in the stone
    // (in the rock's own tile, so it stays put as the camera moves)
    float grain = 0.84 + 0.2 * valueNoise(aGrain * 0.9) + 0.12 * valueNoise(aGrain * 4.3);
    vec3 albedo = aColour * grain;
    albedo = mix(albedo, min(snowColour * 1.4, vec3(1.0)), snowAt(aWorld, n));
    // Faces turned up catch the sky; those turned down sit in shadow
    float sky = 0.75 + 0.35 * dot(n, up);
    vec3 colour = albedo * (ambientLight * 1.4 * sky + sunColour * diffuse * 0.85);

    float radialDistance = length(viewPos - aPos);
    // (the sky's own colour is only the haze until the fog's colour is set: not worked out otherwise)
    vec3 skyColour = fogColour;
    if (fogReady < 0.5) {
        vec3 localSkyDir = transpose(mat3(skyRotation)) * normalize(aPos - viewPos);
        float theta = atan(localSkyDir.x, localSkyDir.z);
        if (theta < 0.0) theta += 2.0 * PI;
        skyColour = texture(skyTexture, vec2(theta / (2.0 * PI), asin(clamp(localSkyDir.y, -1.0, 1.0)) / PI + 0.5)).rgb;
    }
    float fogFactor = smoothstep(0.0, 1.0, clamp((2200.0 - radialDistance) / 1200.0, 0.0, 1.0));
    fragColour = vec4(mix(fogHaze(skyColour), colour, fogFactor), 1.0);
}
