#version 330 core

// Window glass, in world coordinates (planet-centred), drawn relative to the camera

layout (location = 0) in vec3 position;
layout (location = 1) in vec3 normal;
layout (location = 2) in vec2 texCoord;

out vec3 aPos;
out vec3 aNormal;

uniform mat4 viewProjection;
uniform vec3 sphereCamera;

void main() {
    vec3 p = position - sphereCamera;
    aPos = p;
    aNormal = normal;
    gl_Position = viewProjection * vec4(p, 1.0);
}
