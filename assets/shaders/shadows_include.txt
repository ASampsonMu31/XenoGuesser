// Shadows cast by the sun (see ShadowMaps): two maps of how near to the sun the nearest thing is,
// a sharp one close round the player and a coarser one further out, looked up for each point
// to see whether something stands between it and the sun. Shared by every shader of something
// lit by the sun (pulled in by #include "shadows_include": see Shader).
uniform sampler2DShadow shadowMapNear;
uniform sampler2DShadow shadowMapFar;
// Each map's view from the sun, for positions relative to the eyes (as aPos)
uniform mat4 shadowMatrixNear;
uniform mat4 shadowMatrixFar;
// How dark the shadows are (0 none: shadows off, or night), how far each map reaches from the
// eyes, a texel of each map, how far a surface is pushed out against its own shadow (each map),
// and the way to the sun
uniform float shadowStrength;
uniform vec2 shadowReach;
uniform vec2 shadowTexel;
uniform vec2 shadowBias;
uniform vec3 shadowSunDir;
// How much further (units) a surface is pushed out against its own shadow than the maps' own
// bias: zero for most things, more for a vehicle on the move, which slides across the map's
// texels each frame and would otherwise speckle light and dark where it shades itself
uniform float shadowExtraBias;

// The share of one map's light that reaches a point (its place in the map, uvz), softened over
// the texels round it: four reads half a texel apart, each softened by the hardware over its own four
float shadowLookup(sampler2DShadow map, vec3 uvz, float texel) {
    float h = texel * 0.5;
    return 0.25 * (texture(map, vec3(uvz.xy + vec2(-h, -h), uvz.z)) + texture(map, vec3(uvz.xy + vec2(h, -h), uvz.z))
                 + texture(map, vec3(uvz.xy + vec2(-h, h), uvz.z)) + texture(map, vec3(uvz.xy + vec2(h, h), uvz.z)));
}

// The share of one map's light at a point, or -1 where the map doesn't reach it
float shadowIn(sampler2DShadow map, mat4 matrix, vec3 pos, vec3 normal, float texel, float bias) {
    // (pushed a little out from its surface and towards the sun, so it doesn't shade itself)
    float push = bias + shadowExtraBias;
    vec3 p = pos + normal * push + shadowSunDir * push * 0.5;
    vec4 c = matrix * vec4(p, 1.0);
    vec3 uvz = c.xyz * 0.5 + 0.5;
    if (uvz.x <= 0.0 || uvz.x >= 1.0 || uvz.y <= 0.0 || uvz.y >= 1.0 || uvz.z >= 1.0) return -1.0;
    return shadowLookup(map, uvz, texel);
}

// How much of the sun reaches a point (pos, relative to the eyes; normal its surface's, or zero):
// 1 in full sun, less in shadow
float sunShadow(vec3 pos, vec3 normal) {
    if (shadowStrength <= 0.0) return 1.0;
    float d = length(pos);
    if (d >= shadowReach.y) return 1.0;
    float lit = -1.0;
    if (d < shadowReach.x) {
        lit = shadowIn(shadowMapNear, shadowMatrixNear, pos, normal, shadowTexel.x, shadowBias.x);
        // (eased into the far map's towards the edge of the near one)
        float handOver = smoothstep(shadowReach.x * 0.75, shadowReach.x, d);
        if (handOver > 0.0) {
            float far = shadowIn(shadowMapFar, shadowMatrixFar, pos, normal, shadowTexel.y, shadowBias.y);
            if (far >= 0.0 && lit >= 0.0) lit = mix(lit, far, handOver);
        }
    }
    if (lit < 0.0) lit = shadowIn(shadowMapFar, shadowMatrixFar, pos, normal, shadowTexel.y, shadowBias.y);
    if (lit < 0.0) return 1.0;
    // (fading out towards the far map's edge)
    lit = mix(lit, 1.0, smoothstep(shadowReach.y * 0.8, shadowReach.y, d));
    return mix(1.0, lit, shadowStrength);
}
