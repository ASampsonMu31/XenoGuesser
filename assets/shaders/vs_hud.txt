#version 330 core

// Heads-up display: a picture placed on the screen by its rectangle in pixels

layout (location = 0) in vec2 corner;

out vec2 uv;

uniform vec4 rect;        // x, y, width, height in pixels from the top left
uniform vec2 screenSize;

void main() {
    vec2 pixel = rect.xy + corner * rect.zw;
    uv = corner;
    gl_Position = vec4(pixel.x / screenSize.x * 2.0 - 1.0, 1.0 - pixel.y / screenSize.y * 2.0, 0.0, 1.0);
}
