#version 330 core

// Rocks, already placed on the planet, coloured per vertex. Each tile of them is stored
// relative to its own centre and drawn relative to the camera (tileOffset is the tile's
// centre less the camera's position, worked out in double precision; see RockField)

layout (location = 0) in vec3 position;
layout (location = 1) in vec3 normal;
layout (location = 2) in vec3 colour;
// Coarse rocks only: which quarter of their tile (its child cell, 0 to 3) each lies in
layout (location = 3) in float child;

out vec3 aPos;
out vec3 aGrain;
out vec3 aNormal;
out vec3 aColour;
// Planet-centred, for the snow (float is fine for that)
out vec3 aWorld;

uniform mat4 viewProjection;
uniform vec3 tileOffset;
uniform vec3 sphereCamera;
// Coarse rocks in quarters of their tile drawn in detail (bits of detailedChildren) are hidden
uniform int coarse;
uniform int detailedChildren;

void main() {
    vec3 world = position + tileOffset;
    aPos = world;
    aWorld = world + sphereCamera;
    aGrain = position;
    aNormal = normal;
    aColour = colour;
    gl_Position = viewProjection * vec4(world, 1.0);
    if (coarse == 1 && ((detailedChildren >> int(child + 0.5)) & 1) != 0) {
        gl_Position = vec4(2.0, 2.0, 2.0, 1.0);   // outside the view: the detailed rock is here
    }
}
