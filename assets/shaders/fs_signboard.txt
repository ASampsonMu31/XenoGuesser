#version 330 core

// What the distance fades into: an even haze, a greyed version of the sky's colour (greyer
// still under cloud), the same in every direction (see Shader.setFog)
uniform vec3 fogColour;
uniform float fogReady;
vec3 fogHaze(vec3 fallback) {
    return fogReady > 0.5 ? fogColour : fallback;
}

in vec3 aPos;
in vec3 aNormal;
in vec2 aTexCoord;

out vec4 fragColour;

struct Light {
    vec3 position;
    vec3 diffuse;
    vec3 specular;
};

struct Material {
    vec3 ambient;
    vec3 diffuse;
    vec3 specular;
    float shininess;
};

uniform sampler2D skyTexture;
uniform sampler2D alphabetAtlas;

uniform vec3 viewPos;
uniform int numLights;
uniform Light lights[64];
uniform Material material;
uniform vec3 ambientLight;
uniform mat4 skyRotation;

// --- TEXT RENDERING UNIFORMS ---
uniform int textString[512];
uniform int stringLength;
uniform int atlasSize;
uniform int writingDirection; // 0: L->R, 1: R->L, 2: U->D, 3: D->U

// Up to three pictures (the nation's flag, a map, its people, its goods) on the board, which
// the text flows around in reading order
uniform sampler2D picture0;
uniform sampler2D picture1;
uniform sampler2D picture2;
uniform int pictureKind[3];      // 0 none, 1 flag, 2 map, 3 a picture of people or goods
uniform vec4 pictureRect[3];     // x0, y0, x1, y1 as fractions of the text area, y downwards
uniform float pictureAspect[3];  // each picture's width over its height
uniform int pictureFlip[3];      // 1 when its rows run bottom-up
uniform int pictureReady[3];     // 0 while a map is still being drawn
// A title at the beginning of the text, twice the size, in its own lettering
uniform int titleString[16];
uniform int titleLength;
uniform vec4 titleFont;       // ink threshold (weight), slant, width, outline
uniform vec3 titleColour;
uniform int titleShadow;
// A banner shows only the title, in one line of big letters filling the board
uniform int bannerMode;
uniform vec2 areaSize;        // the text area's size, in world units
uniform vec3 boardColour;

const float PI = 3.14159265359;

float glyphInk(int glyph, vec2 uv, float threshold) {
    if (uv.x < 0.0 || uv.x > 1.0 || uv.y < 0.0 || uv.y > 1.0) return 0.0;
    vec4 s = texture(alphabetAtlas, vec2((float(glyph) + clamp(uv.x, 0.02, 0.98)) / float(atlasSize), uv.y));
    float ink = 1.0 - dot(s.rgb, vec3(0.299, 0.587, 0.114));
    return smoothstep(threshold - 0.1, threshold + 0.1, ink);
}

float titleInk(int glyph, vec2 uv) {
    // Width, then slant
    uv.x = (uv.x - 0.5) / titleFont.z + 0.5;
    uv.x += titleFont.y * (0.5 - uv.y);
    float ink = glyphInk(glyph, uv, titleFont.x);
    if (titleFont.w > 0.5) {
        // Outlined: the ink less the ink drawn in from every side
        float d = 0.07;
        float inner = min(min(glyphInk(glyph, uv + vec2(d, 0.0), titleFont.x), glyphInk(glyph, uv - vec2(d, 0.0), titleFont.x)),
                          min(glyphInk(glyph, uv + vec2(0.0, d), titleFont.x), glyphInk(glyph, uv - vec2(0.0, d), titleFont.x)));
        ink = max(0.0, ink - inner);
    }
    return ink;
}

vec3 titleGlyph(vec3 albedo, int glyph, vec2 uv) {
    if (glyph <= 0) return albedo;
    if (titleShadow == 1) {
        albedo = mix(albedo, albedo * 0.45, titleInk(glyph, uv - vec2(0.07, 0.07)) * 0.8);
    }
    return mix(albedo, titleColour, titleInk(glyph, uv));
}

vec3 samplePicture(int i, vec2 st) {
    if (i == 0) return texture(picture0, st).rgb;
    if (i == 1) return texture(picture1, st).rgb;
    return texture(picture2, st).rgb;
}

vec3 pictureBackdrop(int i, vec2 st) {
    if (i == 0) return textureLod(picture0, st, 0.0).rgb;
    if (i == 1) return textureLod(picture1, st, 0.0).rgb;
    return textureLod(picture2, st, 0.0).rgb;
}

void main() {
    vec3 albedo = boardColour;

    if (bannerMode == 1) {
        // A shop's name across its board, as big as fits
        float bx = (aTexCoord.x - 0.03) / 0.94, by = 1.0 - (aTexCoord.y - 0.05) / 0.9;
        if (gl_FrontFacing && bx > 0.0 && bx < 1.0 && by > 0.0 && by < 1.0 && titleLength > 0 && atlasSize > 0) {
            vec2 world = vec2(bx, by) * areaSize;
            bool horizontal = writingDirection < 2;
            float n = float(titleLength);
            // Glyph boxes: tall as the board in a row, wide as it in a column
            float across = horizontal ? areaSize.y : areaSize.x;
            float alongSize = across * 0.85;
            float room = horizontal ? areaSize.x : areaSize.y;
            if (alongSize * n > room * 0.96) { alongSize = room * 0.96 / n; }
            float acrossSize = min(across, alongSize / 0.85);
            float start = (room - alongSize * n) * 0.5;
            float along = horizontal ? world.x : world.y;
            float side = horizontal ? world.y : world.x;
            float offsetAcross = (across - acrossSize) * 0.5;
            float k = (along - start) / alongSize;
            if (k >= 0.0 && k < n && side >= offsetAcross && side <= offsetAcross + acrossSize) {
                int index = int(floor(k));
                // Right to left and bottom to top read from the far end
                if (writingDirection == 1 || writingDirection == 3) index = titleLength - 1 - index;
                vec2 uv = horizontal ? vec2(fract(k), (side - offsetAcross) / acrossSize)
                                     : vec2((side - offsetAcross) / acrossSize, fract(k));
                uv = (uv - 0.06) / 0.88;
                albedo = titleGlyph(albedo, titleString[index], uv);
            }
        }
    } else {

    float marginX = 0.04;
    float marginY = 0.06;
    const float ROWS = 12.0;
    const float LINES = 12.0;

    float mappedX = (aTexCoord.x - marginX) / (1.0 - 2.0 * marginX);
    float mappedY = (aTexCoord.y - marginY) / (1.0 - 2.0 * marginY);
    float invY = 1.0 - mappedY;
    bool inArea = gl_FrontFacing && mappedX > 0.0 && mappedX < 1.0 && invY > 0.0 && invY < 1.0;

    // The grid: horizontal scripts have 28 places on each of 12 lines; vertical ones run
    // 12 places down each of 12 columns, a glyph and a gap wide
    bool horizontal = writingDirection < 2;
    float gridCols = horizontal ? 28.0 : 12.0;
    float places = horizontal ? 28.0 : 12.0;
    float colF = mappedX * gridCols;
    float rowF = invY * ROWS;
    // Reading order: how far along its line, and which line
    float posF = writingDirection == 0 ? colF : writingDirection == 1 ? gridCols - colF
               : writingDirection == 2 ? rowF : ROWS - rowF;
    float lineF = horizontal ? rowF : colF;

    // Each picture's cells, on the board and in reading order
    ivec4 board[3];
    ivec4 reading[3];   // first line, last line + 1, first place, last place + 1
    bool present[3];
    for (int i = 0; i < 3; i++) {
        int c0 = int(floor(pictureRect[i].x * gridCols + 0.5)), c1 = int(floor(pictureRect[i].z * gridCols + 0.5));
        int r0 = int(floor(pictureRect[i].y * ROWS + 0.5)), r1 = int(floor(pictureRect[i].w * ROWS + 0.5));
        present[i] = pictureKind[i] > 0 && c1 > c0 && r1 > r0;
        board[i] = ivec4(c0, r0, c1, r1);
        if (horizontal) {
            reading[i] = ivec4(r0, r1, writingDirection == 0 ? c0 : int(gridCols) - c1, writingDirection == 0 ? c1 : int(gridCols) - c0);
        } else {
            reading[i] = ivec4(c0, c1, writingDirection == 2 ? r0 : int(ROWS) - r1, writingDirection == 2 ? r1 : int(ROWS) - r0);
        }
    }

    // The title takes the first two lines, beside the pictures if there's room for it all
    // there, or else above them, the pictures moving down (or across) out of its way
    int titleLines = 0, titleStart = 0, titleEnd = int(places);
    if (titleLength > 0) {
        for (int i = 0; i < 3; i++) {
            if (present[i] && reading[i].x < 2 && reading[i].y > 0) {
                if (reading[i].z <= 0) titleStart = max(titleStart, reading[i].w); else titleEnd = min(titleEnd, reading[i].z);
            }
        }
        if (titleEnd - titleStart < 2 * titleLength) {
            titleStart = 0;
            titleEnd = int(places);
            for (int i = 0; i < 3; i++) {
                if (present[i] && reading[i].x < 2 && reading[i].y > 0) {
                    int shift = 2 - reading[i].x;
                    reading[i].x += shift;
                    reading[i].y = min(reading[i].y + shift, int(LINES));
                    if (horizontal) { board[i].y = reading[i].x; board[i].w = reading[i].y; }
                    else { board[i].x = reading[i].x; board[i].z = reading[i].y; }
                    present[i] = reading[i].y > reading[i].x;
                }
            }
        }
        titleLines = 2;
    }
    int titleShown = min(titleLength, (titleEnd - titleStart) / 2);

    bool isTextSpace = false;
    int cellIndex = -1;
    float localU = 0.0;
    float localV = 0.0;
    float pad = 0.05;
    int line = int(floor(lineF)), pos = int(floor(posF));

    if (inArea && line >= titleLines && line < int(LINES) && pos >= 0 && pos < int(places)) {
        // Where a glyph sits within its cell; in vertical scripts it takes the left of a wider column
        float colLocalU = horizontal ? fract(colF) : fract(colF) * (28.0 / 12.0);
        float rowLocalV = fract(rowF);
        bool inPicture = false;
        int skipped = 0;
        for (int i = 0; i < 3; i++) {
            if (!present[i]) continue;
            ivec4 r = reading[i];
            int w = r.w - r.z;
            if (line >= r.x && line < r.y && pos >= r.z && pos < r.w) inPicture = true;
            // Count the picture's cells before this one in reading order
            skipped += max(0, min(line, r.y) - max(r.x, titleLines)) * w;
            if (line >= r.x && line < r.y && pos >= r.w) skipped += w;
        }
        if (!inPicture && colLocalU >= pad && colLocalU <= 1.0 - pad && rowLocalV >= pad && rowLocalV <= 1.0 - pad) {
            isTextSpace = true;
            cellIndex = (line - titleLines) * int(places) + pos - skipped;
            localU = (colLocalU - pad) / (1.0 - 2.0 * pad);
            localV = (rowLocalV - pad) / (1.0 - 2.0 * pad);
        }
    }

    // The title: glyphs two places long and two lines deep
    if (inArea && titleLines == 2 && lineF < 2.0 && posF >= float(titleStart) && posF < float(titleStart + 2 * titleShown) && atlasSize > 0) {
        float along = (posF - float(titleStart)) / 2.0;
        int k = int(floor(along));
        float a = fract(along);
        // Across the line: both lines of a horizontal script; in a vertical one, a glyph's width of the two columns
        float b = horizontal ? lineF / 2.0 : mappedX * 14.0;
        vec2 uv = horizontal ? vec2(writingDirection == 0 ? a : 1.0 - a, b) : vec2(b, writingDirection == 2 ? a : 1.0 - a);
        uv = (uv - 0.06) / 0.88;
        albedo = titleGlyph(albedo, titleString[k], uv);
    }

    // The pictures, inset a little from the text, each with a thin dark frame
    for (int i = 0; i < 3; i++) {
        if (!inArea || !present[i]) continue;
        vec2 rmin = vec2(float(board[i].x) / gridCols, float(board[i].y) / ROWS) + vec2(0.3, 0.3) / areaSize;
        vec2 rmax = vec2(float(board[i].z) / gridCols, float(board[i].w) / ROWS) - vec2(0.3, 0.3) / areaSize;
        vec2 q = (vec2(mappedX, invY) - rmin) / (rmax - rmin);
        vec2 sizeW = (rmax - rmin) * areaSize;
        float rectAspect = sizeW.x / sizeW.y;
        vec2 t;
        vec2 frameSize = sizeW;
        if (pictureKind[i] == 2) {
            // A map fills its space, trimmed to fit
            vec2 keep = rectAspect > pictureAspect[i] ? vec2(1.0, pictureAspect[i] / rectAspect) : vec2(rectAspect / pictureAspect[i], 1.0);
            t = (q - 0.5) * keep + 0.5;
        } else {
            // A flag or a picture is shown whole
            vec2 grow = rectAspect > pictureAspect[i] ? vec2(rectAspect / pictureAspect[i], 1.0) : vec2(1.0, pictureAspect[i] / rectAspect);
            t = (q - 0.5) * grow + 0.5;
            if (pictureKind[i] == 1) {
                // The frame hugs the flag itself
                q = t;
                frameSize = sizeW / grow;
            }
        }
        vec2 edge = 0.22 / frameSize;
        if (q.x >= -edge.x && q.x <= 1.0 + edge.x && q.y >= -edge.y && q.y <= 1.0 + edge.y) {
            bool frame = q.x < 0.0 || q.x > 1.0 || q.y < 0.0 || q.y > 1.0;
            vec2 st = vec2(clamp(t.x, 0.0, 1.0), clamp(t.y, 0.0, 1.0));
            if (pictureFlip[i] == 1) st.y = 1.0 - st.y;
            vec3 picture;
            if (pictureReady[i] == 0) {
                picture = vec3(0.86, 0.86, 0.84);
            } else if (t.x < 0.0 || t.x > 1.0 || t.y < 0.0 || t.y > 1.0) {
                // Beside a picture: its backdrop carried on
                picture = pictureBackdrop(i, vec2(0.01, pictureFlip[i] == 1 ? 0.99 : 0.01));
            } else {
                picture = samplePicture(i, st);
            }
            albedo = frame ? vec3(0.15) : picture;
        }
    }

    // Render Character Glyph
    if (isTextSpace && cellIndex >= 0 && cellIndex < stringLength && cellIndex < 512 && atlasSize > 0) {
        int glyphIndex = textString[cellIndex];
        if (glyphIndex > 0) {
            float glyphCoverage = glyphInk(glyphIndex, vec2(localU, localV), 0.5);
            vec3 textInkColor = vec3(0.05, 0.05, 0.05);
            albedo = mix(albedo, textInkColor, glyphCoverage);
        }
    }

    }

    // Standard Lighting Calculations
    vec3 ambient = ambientLight * albedo * 1.0;
    vec3 norm = gl_FrontFacing ? normalize(aNormal) : -normalize(aNormal);
    vec3 viewDir = normalize(viewPos - aPos);
    vec3 totalDiffuse = vec3(0.0);
    vec3 totalSpecular = vec3(0.0);

    for (int i = 0; i < numLights; i++) {
        Light light = lights[i];
        vec3 lightDir = normalize(light.position - aPos);
        float diff = max(dot(norm, lightDir), 0.0);
        totalDiffuse += light.diffuse * albedo * diff;

        vec3 halfwayDir = normalize(lightDir + viewDir);
        float spec = pow(max(dot(norm, halfwayDir), 0.0), material.shininess);
        totalSpecular += light.specular * spec * vec3(0.1);
    }

    vec3 result = ambient + totalDiffuse + totalSpecular;

    // --- SEAMLESS INSIDE-SPHERE FOG COORD MAPPING ---
    float radialDistance = length(viewPos - aPos);
    vec3 worldLookDir = normalize(aPos - viewPos);
    mat3 invSkyRot = transpose(mat3(skyRotation));
    vec3 localSkyDir = invSkyRot * worldLookDir;

    float theta = atan(localSkyDir.x, localSkyDir.z);
    if (theta < 0.0) {
        theta += 2.0 * PI;
    }
    float u = theta / (2.0 * PI);
    float v = asin(clamp(localSkyDir.y, -1.0, 1.0)) / PI + 0.5;

    vec3 dynamicSkyColour = texture(skyTexture, vec2(u, v)).rgb;

    // --- CUBIC DISTANCE BLENDING ---
    float fogStart = 1000.0;
    float fogEnd = 2200.0;

    float fogFactor = (fogEnd - radialDistance) / (fogEnd - fogStart);
    fogFactor = clamp(fogFactor, 0.0, 1.0);
    fogFactor = smoothstep(0.0, 1.0, fogFactor);

    vec3 finalColourWithFog = mix(fogHaze(dynamicSkyColour), result, fogFactor);

    fragColour = vec4(finalColourWithFog, 1.0);
}
