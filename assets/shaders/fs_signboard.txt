#version 330 core

in vec3 aPos;
in vec3 aNormal;
in vec2 aTexCoord;

out vec4 fragColour;

struct Light {
    vec3 position;
    vec3 diffuse;
    vec3 specular;
};

struct Material {
    vec3 ambient;
    vec3 diffuse;
    vec3 specular;
    float shininess;
};

uniform sampler2D skyTexture;
uniform sampler2D alphabetAtlas;

uniform vec3 viewPos;
uniform int numLights;
uniform Light lights[64];
uniform Material material;
uniform vec3 ambientLight;
uniform mat4 skyRotation;

// --- TEXT RENDERING UNIFORMS ---
uniform int textString[512];
uniform int stringLength;
uniform int atlasSize;
uniform int writingDirection; // 0: L->R, 1: R->L, 2: U->D, 3: D->U

// A picture (the nation's flag, a map of the area, or one of its people) somewhere on the
// board, which the text flows around in reading order
uniform sampler2D pictureTexture;
uniform int pictureKind;      // 0 none, 1 flag, 2 map, 3 person
uniform vec4 pictureRect;     // x0, y0, x1, y1 as fractions of the text area, y downwards
uniform float pictureAspect;  // the picture's width over its height
uniform int pictureFlip;      // 1 when its rows run bottom-up
uniform int pictureReady;     // 0 while a map is still being drawn
// A title at the beginning of the text, twice the size, in its own lettering
uniform int titleString[16];
uniform int titleLength;
uniform vec4 titleFont;       // ink threshold (weight), slant, width, outline
uniform vec3 titleColour;
uniform int titleShadow;

const float PI = 3.14159265359;
// The text area's size on the board, in world units
const float AREA_W = 27.6;
const float AREA_H = 19.36;

float glyphInk(int glyph, vec2 uv, float threshold) {
    if (uv.x < 0.0 || uv.x > 1.0 || uv.y < 0.0 || uv.y > 1.0) return 0.0;
    vec4 s = texture(alphabetAtlas, vec2((float(glyph) + clamp(uv.x, 0.02, 0.98)) / float(atlasSize), uv.y));
    float ink = 1.0 - dot(s.rgb, vec3(0.299, 0.587, 0.114));
    return smoothstep(threshold - 0.1, threshold + 0.1, ink);
}

float titleInk(int glyph, vec2 uv) {
    // Width, then slant
    uv.x = (uv.x - 0.5) / titleFont.z + 0.5;
    uv.x += titleFont.y * (0.5 - uv.y);
    float ink = glyphInk(glyph, uv, titleFont.x);
    if (titleFont.w > 0.5) {
        // Outlined: the ink less the ink drawn in from every side
        float d = 0.07;
        float inner = min(min(glyphInk(glyph, uv + vec2(d, 0.0), titleFont.x), glyphInk(glyph, uv - vec2(d, 0.0), titleFont.x)),
                          min(glyphInk(glyph, uv + vec2(0.0, d), titleFont.x), glyphInk(glyph, uv - vec2(0.0, d), titleFont.x)));
        ink = max(0.0, ink - inner);
    }
    return ink;
}

void main() {
    vec3 albedo = vec3(1.0, 1.0, 1.0); // Pure white background

    float marginX = 0.04;
    float marginY = 0.06;
    const float ROWS = 12.0;
    const float LINES = 12.0;

    float mappedX = (aTexCoord.x - marginX) / (1.0 - 2.0 * marginX);
    float mappedY = (aTexCoord.y - marginY) / (1.0 - 2.0 * marginY);
    float invY = 1.0 - mappedY;
    bool inArea = gl_FrontFacing && mappedX > 0.0 && mappedX < 1.0 && invY > 0.0 && invY < 1.0;

    // The grid: horizontal scripts have 28 places on each of 12 lines; vertical ones run
    // 12 places down each of 12 columns, a glyph and a gap wide
    bool horizontal = writingDirection < 2;
    float gridCols = horizontal ? 28.0 : 12.0;
    float places = horizontal ? 28.0 : 12.0;
    float colF = mappedX * gridCols;
    float rowF = invY * ROWS;
    // Reading order: how far along its line, and which line
    float posF = writingDirection == 0 ? colF : writingDirection == 1 ? gridCols - colF
               : writingDirection == 2 ? rowF : ROWS - rowF;
    float lineF = horizontal ? rowF : colF;

    // The picture's cells, on the board and in reading order
    int c0 = int(floor(pictureRect.x * gridCols + 0.5)), c1 = int(floor(pictureRect.z * gridCols + 0.5));
    int r0 = int(floor(pictureRect.y * ROWS + 0.5)), r1 = int(floor(pictureRect.w * ROWS + 0.5));
    bool hasPicture = pictureKind > 0 && c1 > c0 && r1 > r0;
    int l0 = 0, l1 = 0, p0 = 0, p1 = 0;
    if (hasPicture) {
        if (horizontal) {
            l0 = r0; l1 = r1;
            p0 = writingDirection == 0 ? c0 : int(gridCols) - c1;
            p1 = writingDirection == 0 ? c1 : int(gridCols) - c0;
        } else {
            l0 = c0; l1 = c1;
            p0 = writingDirection == 2 ? r0 : int(ROWS) - r1;
            p1 = writingDirection == 2 ? r1 : int(ROWS) - r0;
        }
    }

    // The title takes the first two lines, beside the picture if there's room for it all
    // there, or else above it, the picture moving down (or across) out of its way
    int titleLines = 0, titleStart = 0, titleEnd = int(places);
    if (titleLength > 0) {
        if (hasPicture && l0 < 2 && l1 > 0) {
            int room = p0 <= 0 ? int(places) - p1 : p0;
            if (room >= 2 * titleLength) {
                if (p0 <= 0) titleStart = p1; else titleEnd = p0;
            } else {
                int shift = 2 - l0;
                l0 += shift;
                l1 = min(l1 + shift, int(LINES));
                if (horizontal) { r0 = l0; r1 = l1; } else { c0 = l0; c1 = l1; }
                hasPicture = l1 > l0;
            }
        }
        titleLines = 2;
    }
    int titleShown = min(titleLength, (titleEnd - titleStart) / 2);

    bool isTextSpace = false;
    int cellIndex = -1;
    float localU = 0.0;
    float localV = 0.0;
    float pad = 0.05;
    int line = int(floor(lineF)), pos = int(floor(posF));

    if (inArea && line >= titleLines && line < int(LINES) && pos >= 0 && pos < int(places)) {
        // Where a glyph sits within its cell; in vertical scripts it takes the left of a wider column
        float colLocalU = horizontal ? fract(colF) : fract(colF) * (28.0 / 12.0);
        float rowLocalV = fract(rowF);
        bool inPicture = hasPicture && line >= l0 && line < l1 && pos >= p0 && pos < p1;
        if (!inPicture && colLocalU >= pad && colLocalU <= 1.0 - pad && rowLocalV >= pad && rowLocalV <= 1.0 - pad) {
            isTextSpace = true;
            // Count the cells before this one in reading order, leaving out the picture's
            int w = hasPicture ? p1 - p0 : 0;
            int skippedLines = hasPicture ? max(0, min(line, l1) - max(l0, titleLines)) : 0;
            int skippedHere = (hasPicture && line >= l0 && line < l1 && pos >= p1) ? w : 0;
            cellIndex = (line - titleLines) * int(places) - skippedLines * w + pos - skippedHere;
            localU = (colLocalU - pad) / (1.0 - 2.0 * pad);
            localV = (rowLocalV - pad) / (1.0 - 2.0 * pad);
        }
    }

    // The title: glyphs two places long and two lines deep
    if (inArea && titleLines == 2 && lineF < 2.0 && posF >= float(titleStart) && posF < float(titleStart + 2 * titleShown) && atlasSize > 0) {
        float along = (posF - float(titleStart)) / 2.0;
        int k = int(floor(along));
        float a = fract(along);
        // Across the line: both lines of a horizontal script; in a vertical one, a glyph's width of the two columns
        float b = horizontal ? lineF / 2.0 : mappedX * 14.0;
        vec2 uv = horizontal ? vec2(writingDirection == 0 ? a : 1.0 - a, b) : vec2(b, writingDirection == 2 ? a : 1.0 - a);
        uv = (uv - 0.06) / 0.88;
        int glyph = titleString[k];
        if (glyph > 0) {
            if (titleShadow == 1) {
                albedo = mix(albedo, vec3(0.62), titleInk(glyph, uv - vec2(0.07, 0.07)) * 0.8);
            }
            albedo = mix(albedo, titleColour, titleInk(glyph, uv));
        }
    }

    // The picture, inset a little from the text, with a thin dark frame
    if (inArea && hasPicture) {
        vec2 rmin = vec2(float(c0) / gridCols, float(r0) / ROWS) + vec2(0.3 / AREA_W, 0.3 / AREA_H);
        vec2 rmax = vec2(float(c1) / gridCols, float(r1) / ROWS) - vec2(0.3 / AREA_W, 0.3 / AREA_H);
        vec2 q = (vec2(mappedX, invY) - rmin) / (rmax - rmin);
        vec2 sizeW = (rmax - rmin) * vec2(AREA_W, AREA_H);
        float rectAspect = sizeW.x / sizeW.y;
        vec2 t;
        vec2 frameSize = sizeW;
        if (pictureKind == 2) {
            // A map fills its space, trimmed to fit
            vec2 keep = rectAspect > pictureAspect ? vec2(1.0, pictureAspect / rectAspect) : vec2(rectAspect / pictureAspect, 1.0);
            t = (q - 0.5) * keep + 0.5;
        } else {
            // A flag or a person is shown whole
            vec2 grow = rectAspect > pictureAspect ? vec2(rectAspect / pictureAspect, 1.0) : vec2(1.0, pictureAspect / rectAspect);
            t = (q - 0.5) * grow + 0.5;
            if (pictureKind == 1) {
                // The frame hugs the flag itself
                q = t;
                frameSize = sizeW / grow;
            }
        }
        vec2 edge = 0.22 / frameSize;
        if (q.x >= -edge.x && q.x <= 1.0 + edge.x && q.y >= -edge.y && q.y <= 1.0 + edge.y) {
            bool frame = q.x < 0.0 || q.x > 1.0 || q.y < 0.0 || q.y > 1.0;
            vec2 st = vec2(clamp(t.x, 0.0, 1.0), clamp(t.y, 0.0, 1.0));
            if (pictureFlip == 1) st.y = 1.0 - st.y;
            vec3 picture;
            if (pictureReady == 0) {
                picture = vec3(0.86, 0.86, 0.84);
            } else if (t.x < 0.0 || t.x > 1.0 || t.y < 0.0 || t.y > 1.0) {
                // Beside a person: their backdrop carried on
                picture = textureLod(pictureTexture, vec2(0.01, pictureFlip == 1 ? 0.99 : 0.01), 0.0).rgb;
            } else {
                picture = texture(pictureTexture, st).rgb;
            }
            albedo = frame ? vec3(0.15) : picture;
        }
    }

    // Render Character Glyph
    if (isTextSpace && cellIndex >= 0 && cellIndex < stringLength && cellIndex < 512 && atlasSize > 0) {
        int glyphIndex = textString[cellIndex];
        if (glyphIndex > 0) {
            float glyphCoverage = glyphInk(glyphIndex, vec2(localU, localV), 0.5);
            vec3 textInkColor = vec3(0.05, 0.05, 0.05);
            albedo = mix(albedo, textInkColor, glyphCoverage);
        }
    }

    // Standard Lighting Calculations
    vec3 ambient = ambientLight * albedo * 1.0;
    vec3 norm = gl_FrontFacing ? normalize(aNormal) : -normalize(aNormal);
    vec3 viewDir = normalize(viewPos - aPos);
    vec3 totalDiffuse = vec3(0.0);
    vec3 totalSpecular = vec3(0.0);

    for (int i = 0; i < numLights; i++) {
        Light light = lights[i];
        vec3 lightDir = normalize(light.position - aPos);
        float diff = max(dot(norm, lightDir), 0.0);
        totalDiffuse += light.diffuse * albedo * diff;

        vec3 halfwayDir = normalize(lightDir + viewDir);
        float spec = pow(max(dot(norm, halfwayDir), 0.0), material.shininess);
        totalSpecular += light.specular * spec * vec3(0.1);
    }

    vec3 result = ambient + totalDiffuse + totalSpecular;

    // --- SEAMLESS INSIDE-SPHERE FOG COORD MAPPING ---
    float radialDistance = length(viewPos - aPos);
    vec3 worldLookDir = normalize(aPos - viewPos);
    mat3 invSkyRot = transpose(mat3(skyRotation));
    vec3 localSkyDir = invSkyRot * worldLookDir;

    float theta = atan(localSkyDir.x, localSkyDir.z);
    if (theta < 0.0) {
        theta += 2.0 * PI;
    }
    float u = theta / (2.0 * PI);
    float v = asin(clamp(localSkyDir.y, -1.0, 1.0)) / PI + 0.5;

    vec3 dynamicSkyColour = texture(skyTexture, vec2(u, v)).rgb;

    // --- CUBIC DISTANCE BLENDING ---
    float fogStart = 1000.0;
    float fogEnd = 2200.0;

    float fogFactor = (fogEnd - radialDistance) / (fogEnd - fogStart);
    fogFactor = clamp(fogFactor, 0.0, 1.0);
    fogFactor = smoothstep(0.0, 1.0, fogFactor);

    vec3 finalColourWithFog = mix(dynamicSkyColour, result, fogFactor);

    fragColour = vec4(finalColourWithFog, 1.0);
}
