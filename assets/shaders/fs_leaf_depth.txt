#version 330 core

in vec2 aTexCoord;
uniform sampler2D diffuse_texture;

void main() {
    // Sample the texture for the alpha channel
    vec4 texColor = texture(diffuse_texture, aTexCoord);
    
    // Discard transparent pixels so they don't write to the depth buffer
    if (texColor.a < 0.5) {
        discard;
    }
    
    // Otherwise, do nothing; the fixed-function pipeline handles the depth write
}
