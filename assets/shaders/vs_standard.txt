#version 330 core

layout (location = 0) in vec3 position;
layout (location = 1) in vec3 normal;
layout (location = 2) in vec2 texCoord;

out vec3 aPos;
out vec3 aNormal;
out vec2 aTexCoord;
out float PlanarDepth;

uniform mat4 model;
uniform mat4 mvpMatrix;
// Posters on round walls: the board's edges curve back by bend times the square of the
// distance across it (in the board's own units, -0.5 to 0.5), round a cylinder
uniform float bend;

// The planet's curve (see Planet): lowered by curvature times the square of the distance
// across the ground from curveCentre, along curveDown, the projection of a step up
uniform vec2 curveCentre;
uniform float curvature;
uniform vec4 curveDown;

vec4 planetDrop(vec3 world) {
    vec2 d = world.xz - curveCentre;
    return curveDown * (curvature * dot(d, d));
}

void main() {
    vec3 p = position;
    vec3 n = normal;
    if (bend != 0.0) {
        p.y -= bend * position.x * position.x;
        n = normalize(vec3(2.0 * bend * position.x, 1.0, 0.0));
    }
    gl_Position = mvpMatrix * vec4(p, 1.0) - planetDrop(vec3(model * vec4(p, 1.0)));

    // Exact forward-facing depth used by water screen-space thickness calculations
    PlanarDepth = gl_Position.w;

    aPos = vec3(model * vec4(p, 1.0f));
    aNormal = mat3(transpose(inverse(model))) * n;
    aTexCoord = texCoord;
}
