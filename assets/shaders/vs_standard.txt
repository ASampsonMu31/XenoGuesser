#version 330 core

layout (location = 0) in vec3 position;
layout (location = 1) in vec3 normal;
layout (location = 2) in vec2 texCoord;

out vec3 aPos;
out vec3 aWorldPos;
out vec3 aNormal;
out vec2 aTexCoord;
out float PlanarDepth;

uniform mat4 model;
// The view and projection, camera-relative: the model matrix puts the mesh on the planet
// (planet-centred) and the eyes' place, sphereCamera, is taken off before it, so precision
// holds however far the place is from the planet's centre. A picture of something on its own
// has a view-projection of its own and sphereCamera at the origin.
uniform mat4 viewProjection;
uniform vec3 sphereCamera;
// Posters on round walls: the board's edges curve back by bend times the square of the
// distance across it (in the board's own units, -0.5 to 0.5), round a cylinder
uniform float bend;


void main() {
    vec3 p = position;
    vec3 n = normal;
    if (bend != 0.0) {
        p.y -= bend * position.x * position.x;
        n = normalize(vec3(2.0 * bend * position.x, 1.0, 0.0));
    }
    vec3 world = vec3(model * vec4(p, 1.0));
    aWorldPos = world;
    vec3 worldNormal = mat3(transpose(inverse(model))) * n;
    vec3 eyeRelative = world - sphereCamera;
    gl_Position = viewProjection * vec4(eyeRelative, 1.0);

    // Exact forward-facing depth used by water screen-space thickness calculations
    PlanarDepth = gl_Position.w;

    aPos = eyeRelative;
    aNormal = worldNormal;
    aTexCoord = texCoord;
}
