#version 330 core

// The oxygen's downgrade (M52), and returning to the pod's fizzle: one triangle covering the
// screen, the finished view drawn again through it

out vec2 uv;

void main() {
    vec2 corner = vec2(float((gl_VertexID << 1) & 2), float(gl_VertexID & 2));
    uv = corner;
    gl_Position = vec4(corner * 2.0 - 1.0, 0.0, 1.0);
}
